33/// Check if a tool call is permitted, requesting human approval if needed.
34/// Returns true if the operation should proceed, false if denied.
35pub(crate) async fn check_tool_permission(
36    tool_call: &ToolCall,
37    permissions: &Permissions,
38    pending_approvals: &Option<PendingApprovals>,
39    event_tx: &mpsc::Sender<EngineEvent>,
40    agent_name: &str,
41) -> bool {
42    // Determine the permission type and operation description
43    let (perm_type, operation_desc) = classify_tool_operation(tool_call);
44
45    // Check if explicitly denied
46    let denied = match perm_type.as_str() {
47        "command.run" => check_command_run(permissions).is_err(),
48        "fs.write" => check_fs_write(permissions).is_err(),
49        "fs.read" => check_fs_read(permissions).is_err(),
50        "net.http" => check_net_http(permissions).is_err(),
51        "mcp.use" => check_mcp_use(permissions).is_err(),
52        "skill.use" => check_skill_use(permissions).is_err(),
53        _ => false, // unknown types are allowed by default
54    };
55
56    if denied {
57        let _ = event_tx
58            .send(EngineEvent::Error {
59                agent_id: None,
60                message: format!(
61                    "Agent '{}' attempted denied operation: {} ({})",
62                    agent_name, tool_call.function.name, operation_desc
63                ),
64            })
65            .await;
66        return false;
67    }
68
69    // Check if operation is sensitive and requires human approval
70    if is_sensitive_operation(&perm_type, &operation_desc) {
71        if let Some(approvals) = pending_approvals {
72            let id = uuid::Uuid::new_v4().to_string();
73            let (tx, rx) = tokio::sync::oneshot::channel::<bool>();
74
75            // Store the sender
76            {
77                let mut map = approvals.lock().await;
78                map.insert(id.clone(), tx);
79            }
80
81            // Emit approval required event
82            let _ = event_tx
83                .send(EngineEvent::ApprovalRequired {
84                    id: id.clone(),
85                    operation: format!(
86                        "Agent '{}' wants to run: {} ({})",
87                        agent_name, tool_call.function.name, operation_desc
88                    ),
89                })
90                .await;
91
92            // Wait for user response (with timeout to avoid hanging)
93            match tokio::time::timeout(std::time::Duration::from_secs(120), rx).await {
94                Ok(Ok(true)) => {
95                    let _ = event_tx
96                        .send(EngineEvent::AgentMessage {
97                            agent_id: AgentId::new(),
98                            agent_name: agent_name.to_string(),
99                            message: format!("✓ Approved: {}", operation_desc),
100                        })
101                        .await;
102                    true
103                }
104                Ok(Ok(false)) => {
105                    let _ = event_tx
106                        .send(EngineEvent::AgentMessage {
107                            agent_id: AgentId::new(),
108                            agent_name: agent_name.to_string(),
109                            message: format!("✗ Denied: {}", operation_desc),
110                        })
111                        .await;
112                    false
113                }
114                _ => {
115                    // Timeout or channel closed — deny
116                    let _ = event_tx
117                        .send(EngineEvent::Error {
118                            agent_id: None,
119                            message: format!(
120                                "Approval timed out for: {} ({})",
121                                tool_call.function.name, operation_desc
122                            ),
123                        })
124                        .await;
125                    false
126                }
127            }
128        } else {
129            // No approval mechanism available — deny sensitive operations
130            false
131        }
132    } else {
133        // Non-sensitive operation, allowed by default
134        true
135    }
136}

138/// Classify a tool call into a permission type and human-readable description.
139fn classify_tool_operation(tool_call: &ToolCall) -> (String, String) {
140    let name = &tool_call.function.name;
141    let args: serde_json::Value =
142        serde_json::from_str(&tool_call.function.arguments).unwrap_or_default();
143    let task = args.get("task").and_then(|v| v.as_str()).unwrap_or("");
144
145    match name {
146        n if n == "shell" || n.starts_with("shell_") => {
147            ("command.run".into(), format!("shell: {}", task))
148        }
149        n if n == "read" || n == "grep" || n == "glob" => {
150            ("fs.read".into(), format!("filesystem: {}", task))
151        }
152        n if n == "webfetch" || n == "websearch" => {
153            ("net.http".into(), format!("network: {}", task))
154        }
155        n if n.starts_with("mcp_") => ("mcp.use".into(), format!("mcp: {}", task)),
156        n if n == "lsp_query" => ("command.run".into(), format!("lsp: {}", task)),
157        n if n.contains("write") || n.contains("create") || n.contains("delete") => {
158            ("fs.write".into(), format!("filesystem: {}", task))
159        }
160        n if n.contains("http") || n.contains("fetch") || n.contains("web") => {
161            ("net.http".into(), format!("network: {}", task))
162        }
163        n if n == "filesystem" => {
164            // Inspect the operation to classify read/list (safe) vs write/edit/delete.
165            match crate::filesystem::parse_request(task) {
166                Ok(req) if crate::filesystem::is_write_op(&req.op) => {
167                    ("fs.write".into(), format!("filesystem: {}", task))
168                }
169                _ => ("skill.use".into(), format!("filesystem: {}", task)),
170            }
171        }
172        _ => ("skill.use".into(), format!("{}: {}", name, task)),
173    }
174}

176/// Determine if an operation is sensitive enough to require human approval.
177fn is_sensitive_operation(perm_type: &str, operation_desc: &str) -> bool {
178    match perm_type {
179        // Most shell commands are safe (ls, find, grep, etc.); only
180        // flag destructive/privileged ones for human approval.
181        "command.run" => {
182            let lower = operation_desc.to_lowercase();
183            lower.contains("sudo")
184                || lower.contains("rm -rf")
185                || lower.contains("chmod")
186                || lower.contains("chown")
187                || lower.contains("mkfs")
188                || lower.contains("dd if=")
189                || lower.contains(" >/dev/")
190                || lower.contains("shutdown")
191                || lower.contains("reboot")
192                || lower.contains("poweroff")
193                || lower.contains("passwd")
194        }
195        // Filesystem writes to system paths require approval
196        "fs.write" => {
197            operation_desc.contains("/etc/")
198                || operation_desc.contains("/usr/")
199                || operation_desc.contains("/boot/")
200                || operation_desc.contains("sudo")
201                || operation_desc.contains("rm -rf")
202                || operation_desc.contains("chmod")
203        }
204        // Network operations are not sensitive by default
205        "net.http" => false,
206        // Skill usage is not sensitive
207        "skill.use" => false,
208        _ => false,
209    }
210}

216/// Convert a `Skill` to a `ToolDefinition` that can be passed to an LLM.
217pub(crate) fn skill_to_tool_definition(skill: &Skill) -> ToolDefinition {
218    // All skills take a "task" string describing what to do
219    let input_schema = serde_json::json!({
220        "type": "object",
221        "properties": {
222            "task": {
223                "type": "string",
224                "description": format!(
225                    "The specific task to perform using the '{}' skill. Skill instructions: {}",
226                    skill.name, skill.description
227                )
228            }
229        },
230        "required": ["task"]
231    });
232
233    let mut tool = ToolDefinition {
234        name: skill.name.clone(),
235        description: skill.description.clone(),
236        input_schema,
237    };
238
239    // For the shell skill, append the tool inventory so the agent knows which
240    // modern tools are available without having to ask repeatedly.
241    if skill.name.to_lowercase() == "shell" {
242        tool.description = format!(
243            "{}\n\n{}",
244            skill.description,
245            crate::shell::inventory().to_prompt()
246        );
247    }
248
249    // For the filesystem skill, append the JSON task format so the agent knows
250    // exactly how to structure its requests.
251    if skill.name.to_lowercase() == "filesystem" {
252        tool.description = format!("{}\n\n{}", skill.description, FILESYSTEM_TASK_DOC);
253    }
254
255    tool
256}

312/// Execute a `todo` tool call against the database.
313pub(crate) async fn execute_todo_tool(
314    db: &Option<crate::db::Database>,
315    session_id: Option<Uuid>,
316    tool_call: &ToolCall,
317    event_tx: &mpsc::Sender<EngineEvent>,
318) -> std::result::Result<String, String> {
319    let args: serde_json::Value = serde_json::from_str(&tool_call.function.arguments)
320        .map_err(|e| format!("Failed to parse todo arguments: {e}"))?;
321    let action = args.get("action").and_then(|v| v.as_str()).unwrap_or("");
322
323    let Some(db) = db else {
324        return Err("No database available for todo tool".to_string());
325    };
326    let Some(session_id) = session_id else {
327        return Err("No active session for todo tool".to_string());
328    };
329
330    let result = match action {
331        "add" => {
332            let content = args
333                .get("content")
334                .and_then(|v| v.as_str())
335                .ok_or_else(|| "todo add requires 'content'".to_string())?;
336            let priority = args.get("priority").and_then(|v| v.as_str());
337            let todo = db
338                .add_todo(session_id, content, "pending", priority)
339                .await
340                .map_err(|e| e.to_string())?;
341            Ok(format!("Added todo [{}]: {}", todo.id, todo.content))
342        }
343        "update" => {
344            let id = args
345                .get("id")
346                .and_then(|v| v.as_str())
347                .ok_or_else(|| "todo update requires 'id'".to_string())?;
348            let id = Uuid::parse_str(id).map_err(|e| format!("Invalid todo id: {e}"))?;
349            let content = args.get("content").and_then(|v| v.as_str());
350            let status = args.get("status").and_then(|v| v.as_str());
351            let priority = args.get("priority").and_then(|v| v.as_str());
352            db.update_todo(id, content, status, priority)
353                .await
354                .map_err(|e| e.to_string())?;
355            Ok(format!("Updated todo {id}"))
356        }
357        "delete" => {
358            let id = args
359                .get("id")
360                .and_then(|v| v.as_str())
361                .ok_or_else(|| "todo delete requires 'id'".to_string())?;
362            let id = Uuid::parse_str(id).map_err(|e| format!("Invalid todo id: {e}"))?;
363            db.delete_todo(id).await.map_err(|e| e.to_string())?;
364            Ok(format!("Deleted todo {id}"))
365        }
366        "list" => {
367            let todos = db.list_todos(session_id).await.map_err(|e| e.to_string())?;
368            if todos.is_empty() {
369                Ok("No todos for this session.".to_string())
370            } else {
371                let mut out = String::from("Todos:\n");
372                for t in &todos {
373                    let prio = t.priority.as_deref().unwrap_or("-");
374                    out.push_str(&format!(
375                        "- [{}] ({}) {} — {}\n",
376                        t.status, prio, t.id, t.content
377                    ));
378                }
379                Ok(out)
380            }
381        }
382        _ => Err(format!("Unknown todo action: {action}")),
383    };
384
385    // Emit the updated todo list so the TUI can refresh its sidebar.
386    if let Ok(list) = db.list_todos(session_id).await {
387        let _ = event_tx.send(EngineEvent::TodosUpdated(list)).await;
388    }
389
390    result
391}

423/// Execute a `question` tool call: register a pending question, emit an event
424/// for the TUI to display, and await the user's answer.
425pub(crate) async fn execute_question_tool(
426    pending_questions: &Option<PendingQuestions>,
427    tool_call: &ToolCall,
428    event_tx: &mpsc::Sender<EngineEvent>,
429) -> std::result::Result<String, String> {
430    let args: serde_json::Value = serde_json::from_str(&tool_call.function.arguments)
431        .map_err(|e| format!("Failed to parse question arguments: {e}"))?;
432    let question = args
433        .get("question")
434        .and_then(|v| v.as_str())
435        .ok_or_else(|| "question requires 'question'".to_string())?;
436    let options: Vec<String> = args
437        .get("options")
438        .and_then(|v| v.as_array())
439        .map(|arr| {
440            arr.iter()
441                .filter_map(|v| v.as_str().map(|s| s.to_string()))
442                .collect()
443        })
444        .unwrap_or_default();
445    let recommended = args
446        .get("recommended")
447        .and_then(|v| v.as_str())
448        .map(String::from);
449
450    let Some(pending) = pending_questions else {
451        return Err("No question handler available".to_string());
452    };
453
454    let id = Uuid::new_v4().to_string();
455    let (tx, rx) = tokio::sync::oneshot::channel::<String>();
456    pending.lock().await.insert(id.clone(), tx);
457
458    let _ = event_tx
459        .send(EngineEvent::Question {
460            id: id.clone(),
461            question: question.to_string(),
462            options,
463            recommended,
464        })
465        .await;
466
467    // Await the user's answer (with a generous timeout so the turn can resume).
468    match tokio::time::timeout(std::time::Duration::from_secs(600), rx).await {
469        Ok(Ok(answer)) => Ok(format!("User answered: {answer}")),
470        Ok(Err(_)) => Err("Question channel closed without an answer".to_string()),
471        Err(_) => Err("Question timed out waiting for user answer".to_string()),
472    }
473}

553/// Execute an `apply_patch` tool call.
554///
555/// Parses the batch, validates every path (rejecting traversal), requests a
556/// single approval for the whole batch, and only then applies the changes.
557pub(crate) async fn execute_apply_patch_tool(
558    workspace: &Path,
559    permissions: &Permissions,
560    pending_approvals: &Option<PendingApprovals>,
561    event_tx: &mpsc::Sender<EngineEvent>,
562    agent_name: &str,
563    tool_call: &ToolCall,
564) -> std::result::Result<String, String> {
565    let args: serde_json::Value = serde_json::from_str(&tool_call.function.arguments)
566        .map_err(|e| format!("Failed to parse apply_patch arguments: {e}"))?;
567
568    let json = args
569        .get("operations")
570        .map(|v| v.to_string())
571        .unwrap_or_else(|| tool_call.function.arguments.clone());
572
573    let batch = crate::engine::apply_patch::parse_patch_batch(&json)?;
574
575    // Validate every path before requesting approval or touching the filesystem.
576    for op in &batch.operations {
577        crate::engine::apply_patch::resolve_within_workspace(workspace, &op.path)?;
578    }
579
580    // apply_patch only performs filesystem writes.
581    check_fs_write(permissions).map_err(|e| format!("Permission denied: {e}"))?;
582    // Writes outside the workspace additionally require fs.external.
583    let allow_external = crate::permissions::checker::check_fs_external(permissions).is_ok();
584
585    // Build a human-readable summary of the batch for the approval prompt.
586    let summary = batch
587        .operations
588        .iter()
589        .map(|op| format!("{:?} {}", op.op, op.path))
590        .collect::<Vec<_>>()
591        .join(", ");
592
593    // Request ONE approval for the entire batch. If denied, apply nothing.
594    if !request_batch_approval(pending_approvals, event_tx, agent_name, summary).await {
595        return Err("apply_patch was denied by the user; no changes were applied.".to_string());
596    }
597
598    let results = crate::engine::apply_patch::apply_patch_batch(workspace, &batch, allow_external)?;
599
600    // Emit a unified diff for the TUI diff viewer.
601    let diff_text = crate::engine::apply_patch::batch_to_unified_diff(&batch);
602    let _ = event_tx
603        .send(EngineEvent::DiffAvailable {
604            text: diff_text,
605            title: format!("apply_patch — {}", agent_name),
606        })
607        .await;
608
609    Ok(results.join("\n"))
610}

612/// Execute a tool call against a matching skill and return the result as a string.
613pub(crate) async fn execute_skill_tool(

615    agent_name: &str,
616    tool_call: &ToolCall,
617    event_tx: &mpsc::Sender<EngineEvent>,
618    agent_id: &crate::agent::types::AgentId,
619) -> std::result::Result<String, String> {
620    // Find the skill by name
621    let skill = skills
622        .iter()
623        .find(|s| s.name == tool_call.function.name)
624        .ok_or_else(|| {
625            format!(
626                "Skill '{}' not found in agent's skills",
627                tool_call.function.name
628            )
629        })?;
630
631    // Extract the task argument from the JSON string
632    let args: serde_json::Value = serde_json::from_str(&tool_call.function.arguments)
633        .map_err(|e| format!("Failed to parse tool call arguments: {e}"))?;
634    let task = args.get("task").and_then(|v| v.as_str()).unwrap_or("");
635
636    // Emit tool execution tracing event
637    let _ = event_tx
638        .send(EngineEvent::ToolExecution {
639            agent_id: agent_id.clone(),
640            agent_name: agent_name.to_string(),
641            tool_name: skill.name.clone(),
642            task: task.to_string(),
643        })
644        .await;
645
646    // Execute the tool and capture result
647    let skill_name_lower = skill.name.to_lowercase();
648    let result = if skill_name_lower == "shell" {
649        let result = execute_shell_command(task).await;
650        let prompt = crate::shell::inventory().to_prompt();
651        result.map(|r| format!("{prompt}\n\n{r}"))
652    } else if skill_name_lower.contains("web") || skill_name_lower.contains("research") {
653        execute_web_fetch(task).await
654    } else if skill_name_lower == "filesystem" {
655        execute_filesystem_operation(task).await
656    } else {
657        Ok(format!(
658            r#"Executed skill "{}". Here are the skill instructions:
659
660{}
661
662The task requested was: {}"#,
663            skill.name, skill.instructions, task
664        ))
665    };
666
667    // Emit tool result tracing event
668    let summary = match &result {
669        Ok(r) => truncate_output(r, 120),
670        Err(e) => e.clone(),
671    };
672    let _ = event_tx
673        .send(EngineEvent::ToolResult {
674            agent_id: agent_id.clone(),
675            agent_name: agent_name.to_string(),
676            tool_name: skill.name.clone(),
677            success: result.is_ok(),
678            summary,
679        })
680        .await;
681
682    result
683}

685/// Execute a shell command via `sh -c` with a 60-second timeout.
686///
687/// The `task` is a natural-language description from the LLM (e.g.
688/// "Run the tests: cargo test"). We extract the actual command(s) from it so
689/// that prose containing apostrophes (e.g. "we're") is never passed verbatim
690/// to `sh -c`, which would otherwise fail with an "unexpected EOF" error.
691async fn execute_shell_command(task: &str) -> std::result::Result<String, String> {
692    let command = extract_shell_command(task);
693    if command.is_empty() {
694        return Err(
695            "No shell command found in the task. Please provide a shell command to run.".into(),
696        );
697    }
698
699    // Execute with a POSIX shell (`/bin/sh`) rather than the user's interactive
700    // shell (`$SHELL`, e.g. fish). The commands the LLM generates are bash/POSIX
701    // style (nested double quotes, `bash -c '...'`), which fish cannot parse.
702    let shell = "/bin/sh";
703    let child = tokio::process::Command::new(shell)
704        .arg("-c")
705        .arg(&command)
706        .stdout(std::process::Stdio::piped())
707        .stderr(std::process::Stdio::piped())
708        .spawn()
709        .map_err(|e| format!("Failed to spawn shell command: {e}"))?;
710
711    let output = tokio::time::timeout(std::time::Duration::from_secs(60), child.wait_with_output())
712        .await
713        .map_err(|_| "Command timed out after 60 seconds".to_string())?
714        .map_err(|e| format!("Failed to wait for command: {e}"))?;
715
716    let stdout = String::from_utf8_lossy(&output.stdout).into_owned();
717    let stderr = String::from_utf8_lossy(&output.stderr).into_owned();
718
719    // grep (and similar search tools) exit with code 1 when no lines match.
720    // That is a legitimate "no results" outcome, not a failure.
721    let is_search =
722        command.contains("grep") || command.contains("rg ") || command.contains("find ");
723    if output.status.code() == Some(1) && is_search {
724        return Ok("No results found.".to_string());
725    }
726
727    if output.status.success() {
728        let mut result = String::new();
729        if !stdout.is_empty() {
730            result.push_str(&stdout);
731        }
732        if !stderr.is_empty() {
733            if !result.is_empty() {
734                result.push('\n');
735            }
736            result.push_str(&stderr);
737        }
738        if result.is_empty() {
739            result.push_str("Command completed successfully (no output).");
740        }
741        // Ensure trailing newline for clean display
742        if !result.ends_with('\n') {
743            result.push('\n');
744        }
745        Ok(result)
746    } else {
747        let msg = if !stderr.is_empty() {
748            stderr.trim().to_string()
749        } else if !stdout.is_empty() {
750            stdout.trim().to_string()
751        } else {
752            format!("Command failed with exit code: {:?}", output.status.code())
753        };
754        Err(msg)
755    }
756}

758/// Extract the actual shell command(s) from a natural-language task string.
759///
760/// The LLM typically produces a task like "Run the tests: cargo test" or a
761/// multi-line description followed by indented commands. We keep the command
762/// part (after a colon, or lines that look like shell commands) and drop the
763/// surrounding prose, so natural language is never executed by the shell.
764fn extract_shell_command(task: &str) -> String {
765    let mut commands: Vec<String> = Vec::new();
766    for raw_line in task.lines() {
767        let line = raw_line.trim();
768        if line.is_empty() {
769            continue;
770        }
771        // If the whole line already looks like a shell command, use it as-is.
772        // This avoids splitting on colons inside URLs (e.g. https://...) which
773        // would truncate the command and cause unbalanced quotes / EOF errors.
774        let candidate = if looks_like_shell_command(line) {
775            line
776        } else if let Some(idx) = line.rfind(':') {
777            // "description: command" -> take the part after the last colon.
778            let after = line[idx + 1..].trim();
779            if after.is_empty() { line } else { after }
780        } else {
781            line
782        };
783        if looks_like_shell_command(candidate) {
784            commands.push(candidate.to_string());
785        }
786    }
787    commands.join("\n")
788}

905/// Fetch a URL and return the response body text with a 30-second timeout.
906async fn execute_web_fetch(task: &str) -> std::result::Result<String, String> {
907    let trimmed = task.trim();
908
909    // Find a URL in the task string (prefer https://, fall back to http://)
910    let url = if let Some(start) = trimmed.find("https://") {
911        let rest = &trimmed[start..];
912        let end = rest.find(|c: char| c.is_whitespace()).unwrap_or(rest.len());
913        rest[..end].to_string()
914    } else if let Some(start) = trimmed.find("http://") {
915        let rest = &trimmed[start..];
916        let end = rest.find(|c: char| c.is_whitespace()).unwrap_or(rest.len());
917        rest[..end].to_string()
918    } else {
919        // No explicit URL in the task: fall back to a web search so that
920        // general research tasks (e.g. "what will the weather be tomorrow")
921        // still return useful results instead of failing.
922        return crate::tools::web::web_search(trimmed).await;
923    };
924
925    let response = tokio::time::timeout(std::time::Duration::from_secs(30), reqwest::get(&url))
926        .await
927        .map_err(|_| format!("Request timed out after 30 seconds fetching: {url}"))?
928        .map_err(|e| format!("HTTP request failed for {url}: {e}"))?;
929
930    let status = response.status();
931    if !status.is_success() {
932        return Err(format!("HTTP {} fetching {url}", status.as_u16()));
933    }
934
935    let body = response
936        .text()
937        .await
938        .map_err(|e| format!("Failed to read response body: {e}"))?;
939
940    let mut result = format!("Content from {url}:\n\n");
941    result.push_str(&truncate_output(&body, 10_000));
942
943    Ok(result)
944}

980struct TaskToolArgs {
981    task_id: String,
982    description: String,
983    mode: TaskMode,
984    model: Option<String>,
985    tools: Vec<String>,
986    /// Optional name of a configured subagent type (e.g. "reviewer") used as
987    /// the template for this subagent. When `None`, a dynamic subagent is
988    /// created from the task description.
989    agent: Option<String>,
990}
991
992impl TaskToolArgs {
993    /// Parse the `task` tool arguments from the LLM's JSON string.
994    fn parse(arguments: &str) -> std::result::Result<Self, String> {
995        let args: serde_json::Value = serde_json::from_str(arguments)
996            .map_err(|e| format!("Failed to parse task arguments: {e}"))?;
997        let task_id = args
998            .get("task_id")
999            .and_then(|v| v.as_str())
1000            .map(String::from)
1001            .unwrap_or_else(|| Uuid::new_v4().to_string());
1002        let description = args
1003            .get("description")
1004            .and_then(|v| v.as_str())

1008            .get("mode")
1009            .and_then(|v| v.as_str())
1010            .unwrap_or("foreground")
1011        {
1012            "background" => TaskMode::Background,
1013            _ => TaskMode::Foreground,
1014        };
1015        let model = args.get("model").and_then(|v| v.as_str()).map(String::from);
1016        let tools = args
1017            .get("tools")
1018            .and_then(|v| v.as_array())
1019            .map(|arr| {
1020                arr.iter()
1021                    .filter_map(|v| v.as_str().map(String::from))
1022                    .collect()
1023            })
1024            .unwrap_or_default();
1025        let agent = args.get("agent").and_then(|v| v.as_str()).map(String::from);
1026        Ok(Self {
1027            task_id,
1028            description,
1029            mode,
1030            model,
1031            tools,
1032            agent,
1033        })
1034    }

1129/// Execute a dynamic `task` tool call: delegate to a fresh subagent.
1130///
1131/// In `Foreground` mode the subagent is spawned and awaited inline. In
1132/// `Background` mode the subagent runs in its own tokio task, is registered
1133/// in the shared `JobRegistry`, and emits `EngineEvent::SubagentFinished`
1134/// when it completes.
1135#[allow(clippy::too_many_arguments)]
1136pub(crate) async fn execute_task_tool(
1137    tool_call: &ToolCall,
1138    parent_permissions: &Permissions,
1139    llm_registry: &LlmProviderRegistry,
1140    parent_skills: &[Skill],
1141    event_tx: &mpsc::Sender<EngineEvent>,
1142    usage_tx: &Option<mpsc::Sender<UsageEvent>>,
1143    db: &Option<Database>,
1144    session_id: Option<Uuid>,
1145    history_limit_percent: f64,
1146    retry_config: &RetryConfig,
1147    debug: &Arc<AtomicBool>,
1148    depth: u32,
1149    subagent_depth: u32,
1150    parent_name: &str,
1151    parent_id: &AgentId,
1152    parent_model: &str,
1153    job_registry: &Option<Arc<tokio::sync::Mutex<JobRegistry>>>,
1154    subagent_configs: &[AgentConfig],
1155) -> std::result::Result<String, String> {
1156    let args = TaskToolArgs::parse(&tool_call.function.arguments)?;
1157
1158    // Enforce the delegation depth limit.
1159    if depth >= subagent_depth {
1160        return Err(format!(
1161            "subagent depth limit reached ({depth} >= {subagent_depth}): cannot delegate further"
1162        ));
1163    }
1164
1165    // Build the `SpawnSubagentConfig` either from a configured subagent type
1166    // (when `args.agent` is provided) or dynamically from the task description.
1167    let sub_cfg = if let Some(agent_name) = &args.agent {
1168        // Resolve the configured subagent type by name. When found, the
1169        // subagent inherits ALL of that type's instructions: description /
1170        // system prompt, skills, MCPs, model and permissions.
1171        let config = subagent_configs
1172            .iter()
1173            .find(|c| c.name == *agent_name)
1174            .ok_or_else(|| {
1175                let available = subagent_configs
1176                    .iter()
1177                    .map(|c| c.name.as_str())
1178                    .collect::<Vec<_>>()
1179                    .join(", ");
1180                format!("Configured subagent type '{agent_name}' not found. Available: {available}")
1181            })?;
1182
1183        // Derive the child's effective permissions: parent ∩ configured child.
1184        let child_permissions =
1185            parent_permissions.intersection(&Permissions::from_config(&config.permissions));
1186
1187        SpawnSubagentConfig {
1188            config: config.clone(),
1189            parent_id: parent_id.clone(),
1190            llm_registry: llm_registry.clone(),
1191            task: args.description.clone(),
1192            db: db.clone(),
1193            session_id,
1194            event_tx: event_tx.clone(),
1195            usage_tx: usage_tx.clone(),
1196            history_limit_percent,
1197            retry_config: retry_config.clone(),
1198            debug: debug.clone(),
1199            max_steps: config.max_steps,
1200            depth: depth + 1,
1201            // Load the configured type's own skills from `config.skills`.
1202            skills_override: None,
1203            permissions_override: Some(child_permissions),
1204            agent_type: args.agent.clone(),
1205            mode: args.mode.clone(),
1206        }
1207    } else {
1208        // Derive the child's permissions: parent ∩ child (deny propagates down).
1209        //
1210        // NOTE: The `tools` argument does NOT restrict the child's permissions.
1211        // The child's effective permissions are always `parent ∩ child` — the
1212        // parent's permissions intersected with the child's own (here the
1213        // default, i.e. allow-by-default). The `tools` list only filters which
1214        // *skills* the child is granted (see `skills_override` below); it is a
1215        // capability allow-list for skills, not a permission restriction.
1216        // Reimplementing per-tool permission filtering is intentionally out of
1217        // scope here.
1218        let child_permissions = parent_permissions.intersection(&Permissions::default());
1219
1220        // Filter the parent's skills by the requested tool names (or grant all).
1221        // This is the ONLY effect of the `tools` list: it narrows the set of
1222        // skills the child can invoke. It does not alter the child's permissions.
1223        let skills_override: Vec<Skill> = if args.tools.is_empty() {
1224            parent_skills.to_vec()
1225        } else {
1226            parent_skills
1227                .iter()
1228                .filter(|s| args.tools.iter().any(|t| t == &s.name))
1229                .cloned()
1230                .collect()
1231        };
1232
1233        let model = args
1234            .model
1235            .clone()
1236            .unwrap_or_else(|| parent_model.to_string());
1237
1238        let config = AgentConfig {
1239            name: args.task_id.clone(),
1240            description: args.description.clone(),
1241            role: AgentRole::SubAgent,
1242            model,
1243            skills: Vec::new(), // skills provided via `skills_override`
1244            mcps: Vec::new(),
1245            permissions: crate::config::PermissionConfig::default(),
1246            subagents: Vec::new(),
1247            system_prompt: args.description.clone(),
1248            max_steps: 90,
1249            subagent_depth,
1250        };
1251
1252        SpawnSubagentConfig {
1253            config,
1254            parent_id: parent_id.clone(),
1255            llm_registry: llm_registry.clone(),
1256            task: args.description.clone(),
1257            db: db.clone(),
1258            session_id,
1259            event_tx: event_tx.clone(),
1260            usage_tx: usage_tx.clone(),
1261            history_limit_percent,
1262            retry_config: retry_config.clone(),
1263            debug: debug.clone(),
1264            max_steps: 90,
1265            depth: depth + 1,
1266            skills_override: Some(skills_override),
1267            permissions_override: Some(child_permissions),
1268            agent_type: None,
1269            mode: args.mode.clone(),
1270        }
1271    };
1272
1273    match args.mode {
1274        TaskMode::Foreground => {
1275            let _ = event_tx
1276                .send(EngineEvent::AgentStatusChanged {
1277                    agent_id: parent_id.clone(),
1278                    agent_name: parent_name.to_string(),
1279                    status: AgentStatus::WaitingForSubAgent,
1280                })
1281                .await;
1282            let result = spawn_subagent_and_delegate(sub_cfg).await;
1283            let _ = event_tx
1284                .send(EngineEvent::AgentStatusChanged {
1285                    agent_id: parent_id.clone(),
1286                    agent_name: parent_name.to_string(),
1287                    status: AgentStatus::Working,
1288                })
1289                .await;
1290            match result {
1291                Ok(r) => Ok(r),
1292                Err(e) => Err(format!("Subagent error: {e}")),
1293            }
1294        }
1295        TaskMode::Background => {
1296            let task_id = args.task_id.clone();
1297            let task_id_for_register = task_id.clone();
1298            let job_registry_owned = job_registry.clone();
1299            let event_tx_owned = event_tx.clone();
1300            let handle = tokio::spawn(async move {
1301                let summary = match spawn_subagent_and_delegate(sub_cfg).await {
1302                    Ok(r) => r,
1303                    Err(e) => format!("Subagent error: {e}"),
1304                };
1305                let _ = event_tx_owned
1306                    .send(EngineEvent::SubagentFinished {
1307                        task_id: task_id.clone(),
1308                        summary,
1309                    })
1310                    .await;
1311                if let Some(reg) = job_registry_owned {
1312                    reg.lock().await.remove(&task_id);
1313                }
1314            });
1315            if let Some(reg) = job_registry {
1316                reg.lock()
1317                    .await
1318                    .register(task_id_for_register.clone(), handle);
1319            }
1320            // NOTE: When `job_registry` is `None` (e.g. headless tests or a
1321            // caller that did not provide a registry), the `JoinHandle` is
1322            // dropped here and the background task is detached: it keeps
1323            // running and emits `SubagentFinished`, but it is not tracked by
1324            // any `JobRegistry` and therefore never appears in `/jobs`. In the
1325            // engine this is always `Some`, so background jobs are tracked in
1326            // practice; the `None` case is only a documented fallback.
1327            Ok(format!(
1328                "Background task '{task_id_for_register}' launched. It will run asynchronously."
1329            ))
1330        }
1331    }
1332}

1334/// Resolve an LLM provider for a given model name using the registry.
1335fn resolve_provider_for_model(
1336    model: &str,
1337    registry: &LlmProviderRegistry,
1338) -> std::result::Result<Arc<dyn LlmProvider>, String> {
1339    let provider_name = if model.contains('/') {
1340        "openrouter"
1341    } else if model.starts_with("claude") {
1342        "anthropic"
1343    } else if model.starts_with("gpt") || model.starts_with("o1") || model.starts_with("o3") {
1344        "openai"
1345    } else {
1346        "ollama"
1347    };
1348
1349    registry.get(provider_name).ok_or_else(|| {
1350        format!("No provider configured for model '{model}'. Tried provider: {provider_name}")
1351    })
1352}

1383/// Spawn a subagent on-demand, delegate a task to it, and wait for the response.
1384///
1385/// The subagent is created fresh with its own LLM provider and skills.
1386/// It processes the task using its own model (which may differ from the parent's)
1387/// and returns the text response. The subagent task is destroyed after responding.
1388pub(crate) async fn spawn_subagent_and_delegate(cfg: SpawnSubagentConfig) -> Result<String> {
1389    let SpawnSubagentConfig {
1390        config,
1391        parent_id,
1392        llm_registry,
1393        task,
1394        db: _db,
1395        session_id: _session_id,
1396        event_tx,
1397        usage_tx,
1398        history_limit_percent,
1399        retry_config,
1400        debug,
1401        max_steps,
1402        depth: _depth,
1403        skills_override,
1404        permissions_override,
1405        agent_type,
1406        mode,
1407    } = cfg;
1408    // Create agent with SubAgent role
1409    let mut agent = Agent::from_config(&config, AgentRole::SubAgent);
1410    if let Some(perms) = permissions_override {
1411        agent.permissions = perms;
1412    }
1413    let agent_id = agent.id.clone();
1414    let agent_name = agent.name.clone();
1415    let model = agent.model.clone();
1416
1417    // Emit subagent created event
1418    let _ = event_tx
1419        .send(EngineEvent::SubagentCreated {
1420            parent_id: parent_id.clone(),
1421            subagent_id: agent_id.clone(),
1422            subagent_name: agent_name.clone(),
1423            skills: agent
1424                .skills
1425                .iter()
1426                .filter_map(|p| {
1427                    p.file_stem()
1428                        .and_then(|f| f.to_str().map(|s| s.to_string()))
1429                })
1430                .collect(),
1431            mcps: agent.mcps.clone(),
1432            agent_type,
1433            mode,
1434        })
1435        .await;
1436
1437    // Resolve provider
1438    let provider = resolve_provider_for_model(&model, &llm_registry).map_err(Error::Provider)?;
1439
1440    // Load subagent's own skills (or use pre-loaded skills from `skills_override`).
1441    let skills = match skills_override {
1442        Some(s) => s,
1443        None => load_agent_skills(&agent.skills),
1444    };
1445    let subagent_tools: Vec<ToolDefinition> = skills.iter().map(skill_to_tool_definition).collect();
1446
1447    // Load subagent description as system prompt
1448    let system_prompt = agent.description.clone();
1449
1450    // Owned copies for the spawned task
1451    let task_owned = task.to_string();
1452
1453    // Create oneshot channel for the response
1454    let (response_tx, response_rx) = tokio::sync::oneshot::channel::<String>();
1455
1456    // Spawn subagent task
1457    let subagent_history_limit = history_limit_percent;
1458    let subagent_retry_config = retry_config.clone();
1459    let debug_mode = debug;
1460    tokio::spawn(async move {
1461        let mut conversation: Vec<LlmMessage> = Vec::new();
1462
1463        // Context window budget for subagent
1464        let max_history_tokens =
1465            (provider.context_window() as f64 * subagent_history_limit / 100.0) as usize;
1466
1467        // Prepend system prompt if available
1468        if !system_prompt.is_empty() {
1469            conversation.push(LlmMessage {
1470                role: MessageRole::System,
1471                content: system_prompt.clone(),
1472                tool_calls: None,
1473                tool_call_id: None,
1474            });
1475        }
1476
1477        // Add the user task
1478        conversation.push(LlmMessage {
1479            role: MessageRole::User,
1480            content: task_owned,
1481            tool_calls: None,
1482            tool_call_id: None,
1483        });
1484
1485        // Tool loop for subagent (skills only, no nested subagents)
1486        let mut step_count: u32 = 0;
1487        'tool_loop: loop {
1488            step_count += 1;
1489            if step_count > max_steps {
1490                let _ = response_tx.send(format!(
1491                    "[Tarea incompleta] Se alcanzó el límite de {max_steps} pasos (turnos) sin completar la tarea."
1492                ));
1493                break 'tool_loop;
1494            }
1495            // Trim conversation to fit context window
1496            summarize_conversation(
1497                &mut conversation,
1498                max_history_tokens,
1499                Some(&*provider),
1500                &model,
1501                false,
1502                None, // subagents have no access to the parent's tool store
1503            )
1504            .await;
1505
1506            let request = LlmRequest {
1507                model: model.clone(),
1508                messages: conversation.clone(),
1509                tools: subagent_tools.clone(),
1510                max_tokens: None,
1511                temperature: None,
1512                stream: false, // non-streaming for subagents
1513                cache_control: None,
1514            };
1515
1516            // Emit debug event for subagent LLM request if debug mode is on
1517            if debug_mode.load(Ordering::Relaxed) {
1518                let payload =
1519                    serde_json::to_string_pretty(&request).unwrap_or_else(|_| "{}".to_string());
1520                let _ = event_tx
1521                    .send(EngineEvent::LlmRequestDebug {
1522                        agent_name: agent_name.clone(),
1523                        model: model.clone(),
1524                        payload,
1525                    })
1526                    .await;
1527            }
1528
1529            // Wrap subagent LLM call with retries
1530            let sub_retry_cfg = subagent_retry_config.clone();
1531            let sub_agent_name = agent_name.clone();
1532            let complete_result = retry_with_backoff(
1533                |_attempt| {
1534                    let req = request.clone();
1535                    let prov = provider.clone();
1536                    async move { prov.complete(req).await }
1537                },
1538                &sub_retry_cfg,
1539                &format!("Subagent LLM call for '{}'", sub_agent_name),
1540            )
1541            .await;
1542
1543            match complete_result {
1544                Ok(response) => {
1545                    // Emit token usage if available
1546                    if let Some(ref usage) = response.usage {
1547                        let cost = (usage.prompt_tokens as f64
1548                            * provider.input_price_per_million()
1549                            + usage.completion_tokens as f64 * provider.output_price_per_million())
1550                            / 1_000_000.0;
1551                        let _ = event_tx
1552                            .send(EngineEvent::TokenUsage {
1553                                agent_id: agent_id.clone(),
1554                                agent_name: agent_name.clone(),
1555                                total_tokens: usage.total_tokens,
1556                                context_window: provider.context_window() as u32,
1557                                cost,
1558                            })
1559                            .await;
1560                        if let Some(ref utx) = usage_tx {
1561                            let _ = utx
1562                                .send(UsageEvent {
1563                                    total_tokens: usage.total_tokens,
1564                                    cost,
1565                                })
1566                                .await;
1567                        }
1568                    }
1569
1570                    // Emit debug event for subagent LLM response
1571                    if debug_mode.load(Ordering::Relaxed) {
1572                        let resp = LlmResponse {
1573                            content: response.content.clone(),
1574                            tool_calls: response.tool_calls.clone(),
1575                            finish_reason: if response.tool_calls.is_empty() {
1576                                "stop".to_string()
1577                            } else {
1578                                "tool_use".to_string()
1579                            },
1580                            usage: response.usage,
1581                            thinking: None,
1582                        };
1583                        let payload = serde_json::to_string_pretty(&resp)
1584                            .unwrap_or_else(|_| "{}".to_string());
1585                        let _ = event_tx
1586                            .send(EngineEvent::LlmResponseDebug {
1587                                agent_name: agent_name.clone(),
1588                                model: model.clone(),
1589                                payload,
1590                            })
1591                            .await;
1592                    }
1593
1594                    if response.tool_calls.is_empty() {
1595                        // No tool calls — final response
1596                        let _ = response_tx.send(response.content);
1597                        break 'tool_loop;
1598                    }
1599
1600                    // Save assistant message with tool calls
1601                    conversation.push(LlmMessage {
1602                        role: MessageRole::Assistant,
1603                        content: response.content,
1604                        tool_calls: Some(response.tool_calls.clone()),
1605                        tool_call_id: None,
1606                    });
1607
1608                    // Execute each tool call
1609                    for tc in &response.tool_calls {
1610                        let result =
1611                            execute_skill_tool(&skills, &agent_name, tc, &event_tx, &agent_id)
1612                                .await
1613                                .unwrap_or_else(|e| e);
1614
1615                        conversation.push(LlmMessage {
1616                            role: MessageRole::Tool,
1617                            content: result,
1618                            tool_calls: None,
1619                            tool_call_id: Some(tc.id.clone()),
1620                        });
1621                    }
1622                    // Loop back: LLM now has tool results
1623                }
1624                Err(e) => {
1625                    let _ = response_tx.send(format!("Subagent error: {e}"));
1626                    break 'tool_loop;
1627                }
1628            }
1629        }
1630
1631        // Emit subagent completed event
1632        let _ = event_tx
1633            .send(EngineEvent::SubagentCompleted {
1634                subagent_id: agent_id,
1635                subagent_name: agent_name,
1636                result: "completed".into(),
1637            })
1638            .await;
1639    });
1640
1641    // Wait for the response
1642    let response = response_rx
1643        .await
1644        .map_err(|_| Error::Agent("Subagent response channel closed".into()))?;
1645
1646    Ok(response)
1647}
