5/// A typed source of context that can be loaded and refreshed.
6///
7/// `Source<A>` abstracts over any external context that an agent consumes
8/// (files, databases, remote documents, ...). The associated type `A` is the
9/// loaded value produced by [`Source::load`] / [`Source::refresh`].
10pub trait Source<A> {
11    /// Load the source and return its current value.
12    fn load(&self) -> Result<A>;
13
14    /// Re-read the source and return its (possibly updated) value.
15    fn refresh(&mut self) -> Result<A>;
16}
17
18/// A [`Source`] backed by a plain text file on disk.
19///
20/// `load` reads the file contents as a `String`. `refresh` re-reads the file,
21/// so callers can pick up edits made after the initial load.
22#[derive(Debug, Clone)]
23pub struct FileSource {
24    /// Absolute path to the file to read.
25    pub path: PathBuf,
26}
27
28impl FileSource {
29    /// Create a new [`FileSource`] for the given path.
30    pub fn new(path: impl Into<PathBuf>) -> Self {
31        Self { path: path.into() }
32    }
33}
34
35impl Source<String> for FileSource {
36    fn load(&self) -> Result<String> {
37        std::fs::read_to_string(&self.path).map_err(|e| {
38            Error::Io(std::io::Error::new(
39                e.kind(),
40                format!("Failed to read source file '{}': {e}", self.path.display()),
41            ))
42        })
43    }
44
45    fn refresh(&mut self) -> Result<String> {
46        self.load()
47    }
48}

50/// The set of workspace instruction files that are loaded as initial context.
51///
52/// These are conventional files that describe how to work in a repository.
53/// They are detected by name and injected as System messages when present.
54pub const WORKSPACE_INSTRUCTION_FILES: [&str; 3] = ["AGENTS.md", "CLAUDE.md", "CONTEXT.md"];
55
56/// Load the workspace instruction files that exist under `workspace`.
57///
58/// Returns a list of `(file_name, contents)` pairs for every instruction file
59/// found in the workspace root. Files that do not exist are skipped silently.
60pub fn load_workspace_instructions(workspace: &Path) -> Vec<(String, String)> {
61    let mut loaded = Vec::new();
62    for name in WORKSPACE_INSTRUCTION_FILES {
63        let path = workspace.join(name);
64        if path.is_file() {
65            if let Ok(content) = std::fs::read_to_string(&path) {
66                loaded.push((name.to_string(), content));
67            }
68        }
69    }
70    loaded
71}

78    fn file_source_loads_content() {
79        let dir = tempfile::tempdir().expect("tempdir");
80        let path = dir.path().join("note.txt");
81        std::fs::write(&path, "hello world").expect("write");
82        let source = FileSource::new(&path);
83        assert_eq!(source.load().expect("load"), "hello world");
84    }

88        let dir = tempfile::tempdir().expect("tempdir");
89        let path = dir.path().join("note.txt");
90        std::fs::write(&path, "v1").expect("write");
91        let mut source = FileSource::new(&path);
92        assert_eq!(source.load().expect("load"), "v1");
93        std::fs::write(&path, "v2").expect("write");
94        assert_eq!(source.refresh().expect("refresh"), "v2");
95    }
