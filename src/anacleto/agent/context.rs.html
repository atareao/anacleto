5/// Roughly estimate the number of tokens in a text string.
6/// Uses the rule of thumb: ~4 characters per token for English text.
7fn estimate_tokens(text: &str) -> usize {
8    if text.is_empty() {
9        0
10    } else {
11        text.len().div_ceil(4)
12    }
13}

15/// Fraction of the token budget at which compaction is triggered automatically.
16///
17/// When the conversation's estimated token count exceeds
18/// `max_tokens * COMPACTION_THRESHOLD_RATIO`, compaction runs even without an
19/// explicit `/compact` command.
20pub const COMPACTION_THRESHOLD_RATIO: f64 = 0.8;
21
22/// Whether the conversation should be compacted based on the token threshold.
23///
24/// Returns `true` when `total_tokens` exceeds `max_tokens * ratio`. A `ratio`
25/// of `1.0` (or greater) effectively disables threshold-based compaction.
26pub fn should_compact(total_tokens: usize, max_tokens: usize, ratio: f64) -> bool {
27    if max_tokens == 0 {
28        return false;
29    }
30    let threshold = (max_tokens as f64 * ratio) as usize;
31    total_tokens > threshold
32}

38/// The structured summary template used to guide the LLM during compaction.
39///
40/// The LLM is asked to fill only these sections, in order, producing an
41/// anchored, structured summary rather than a free-form blob of text.
42pub const SUMMARY_TEMPLATE: &str = "\
43Resumen estructurado de la conversación. Rellena SOLO estas secciones, en este orden, sin añadir nada más:
44## Objetivo
45## Decisiones tomadas
46## Hechos y contexto clave
47## Código/patrones relevantes
48## Tareas pendientes
49## Riesgos o bloqueos";

51/// Build the summarization prompt for a set of messages to summarize.
52///
53/// The prompt instructs the LLM to produce a structured, anchored summary
54/// following [`SUMMARY_TEMPLATE`], preserving key facts, decisions, code
55/// patterns and context.
56///
57/// When `tool_store` is provided, Tool messages whose `tool_call_id` is present
58/// in the store use the FULL tool output (retained by the [`ToolOutputStore`])
59/// instead of the truncated version the LLM originally received, so the summary
60/// is built from complete information rather than the shortened context.
61fn build_summary_prompt(messages: &[LlmMessage], tool_store: Option<&ToolOutputStore>) -> String {
62    let summary_text: String = messages
63        .iter()
64        .map(|m| {
65            let role_label = match m.role {
66                MessageRole::User => "User",
67                MessageRole::Assistant => "Assistant",
68                MessageRole::Tool => "Tool",
69                MessageRole::System => "System",
70            };
71            // For Tool messages, prefer the full output retained in the store
72            // over the truncated version the LLM originally received.
73            let content = if m.role == MessageRole::Tool {
74                if let (Some(id), Some(store)) = (&m.tool_call_id, tool_store) {
75                    store.get(id).map(|s| s.as_str()).unwrap_or(&m.content)
76                } else {
77                    &m.content
78                }
79            } else {
80                &m.content
81            };
82            format!("{}: {}", role_label, content)
83        })
84        .collect::<Vec<_>>()
85        .join("\n\n");
86
87    format!("{SUMMARY_TEMPLATE}\n\nConversación a resumir:\n\n{summary_text}")
88}

90/// Trim conversation history to fit within a token budget.
91///
92/// Always preserves:
93/// - System messages (role == System)
94/// - The most recent non-system messages (at least the last exchange)
95///
96/// Removes oldest non-system messages first when over budget.
97fn trim_conversation(messages: &mut Vec<LlmMessage>, max_tokens: usize) {
98    if messages.is_empty() || max_tokens == 0 {
99        return;
100    }
101
102    // Quick check: if already under budget, do nothing
103    let total: usize = messages.iter().map(|m| estimate_tokens(&m.content)).sum();
104    if total <= max_tokens {
105        return;
106    }
107
108    // Separate system messages from the rest
109    let mut system_msgs: Vec<LlmMessage> = Vec::new();
110    let mut non_system_msgs: Vec<LlmMessage> = Vec::new();
111
112    for msg in messages.drain(..) {
113        if msg.role == MessageRole::System {
114            system_msgs.push(msg);
115        } else {
116            non_system_msgs.push(msg);
117        }
118    }
119
120    // Remove oldest non-system messages until under budget
121    // Always keep at least the last 2 messages (one exchange)
122    while non_system_msgs.len() > 2 {
123        // Build a temporary reference list to check token count
124        let test_total: usize = system_msgs
125            .iter()
126            .chain(non_system_msgs.iter())
127            .map(|m| estimate_tokens(&m.content))
128            .sum();
129
130        if test_total <= max_tokens {
131            break;
132        }
133        non_system_msgs.remove(0);
134    }
135
136    // Restore: system first, then remaining non-system
137    *messages = system_msgs;
138    messages.extend(non_system_msgs);
139}

141/// Try to summarize old conversation messages using the LLM.
142///
143/// When the conversation exceeds the token budget, this function attempts to
144/// summarize older messages (keeping the latest exchange intact) by calling
145/// the LLM with a summarization prompt. The summary is injected as a System
146/// message. Falls back to `trim_conversation` if the LLM call fails or if
147/// there aren't enough messages to make summarization worthwhile.
148///
149/// When `tool_store` is provided, Tool messages being summarized use the full
150/// output retained in the store (see [`build_summary_prompt`]).
151///
152/// When `force` is `true`, the summarization is attempted even if the
153/// conversation is under the token budget (used by the `/compact` command).
154/// The rest of the logic (needs a provider, needs at least 4 non-system
155/// messages, preserves system + latest exchange) is still respected.
156pub(crate) async fn summarize_conversation(
157    conversation: &mut Vec<LlmMessage>,
158    max_tokens: usize,
159    provider: Option<&dyn LlmProvider>,
160    model: &str,
161    force: bool,
162    tool_store: Option<&ToolOutputStore>,
163) {
164    // Quick check: if already under the compaction threshold, do nothing
165    // (unless forced). The threshold is `max_tokens * COMPACTION_THRESHOLD_RATIO`.
166    let total: usize = conversation
167        .iter()
168        .map(|m| estimate_tokens(&m.content))
169        .sum();
170    if !force && !should_compact(total, max_tokens, COMPACTION_THRESHOLD_RATIO) {
171        return;
172    }
173
174    // Need a provider for summarization
175    let Some(prov) = provider else {
176        trim_conversation(conversation, max_tokens);
177        return;
178    };
179
180    // Separate system from non-system
181    let mut system_msgs: Vec<LlmMessage> = Vec::new();
182    let mut non_system_msgs: Vec<LlmMessage> = Vec::new();
183
184    for msg in conversation.drain(..) {
185        if msg.role == MessageRole::System {
186            system_msgs.push(msg);
187        } else {
188            non_system_msgs.push(msg);
189        }
190    }
191
192    // Need at least 4 non-system messages to make summarization worthwhile
193    if non_system_msgs.len() < 4 {
194        system_msgs.extend(non_system_msgs);
195        *conversation = system_msgs;
196        trim_conversation(conversation, max_tokens);
197        return;
198    }
199
200    // Keep the last 2 messages (latest exchange), summarize the rest
201    let keep = 2;
202    let split_at = non_system_msgs.len() - keep;
203    let to_summarize: Vec<LlmMessage> = non_system_msgs.drain(..split_at).collect();
204    let recent = non_system_msgs;
205
206    // Build summarization prompt using the structured, anchored template.
207    let summary_prompt = build_summary_prompt(&to_summarize, tool_store);
208
209    // Call LLM for summarization
210    let request = LlmRequest {
211        model: model.to_string(),
212        messages: vec![LlmMessage {
213            role: MessageRole::User,
214            content: summary_prompt,
215            tool_calls: None,
216            tool_call_id: None,
217        }],
218        tools: vec![],
219        max_tokens: Some(1024),
220        temperature: Some(0.3),
221        stream: false,
222        cache_control: None,
223    };
224
225    match prov.complete(request).await {
226        Ok(response) => {
227            // Replace summarized messages with an anchored summary system message
228            system_msgs.push(LlmMessage {
229                role: MessageRole::System,
230                content: format!("{SUMMARY_ANCHOR_MARKER}\n{}", response.content.trim()),
231                tool_calls: None,
232                tool_call_id: None,
233            });
234        }
235        Err(_) => {
236            // Fallback: restore and truncate
237            system_msgs.extend(to_summarize);
238            system_msgs.extend(recent);
239            *conversation = system_msgs;
240            trim_conversation(conversation, max_tokens);
241            return;
242        }
243    }
244
245    // Rebuild conversation
246    *conversation = system_msgs;
247    conversation.extend(recent);
248
249    // Final trim check (summary itself might still be over budget)
250    trim_conversation(conversation, max_tokens);
251}

324    #[test]
325    fn test_should_compact_threshold_logic() {
326        let max_tokens = 1000;
327        // Below the 0.8 ratio -> no compaction.
328        assert!(!should_compact(799, max_tokens, COMPACTION_THRESHOLD_RATIO));
329        // Exactly at the threshold (800) -> still not over it.
330        assert!(!should_compact(800, max_tokens, COMPACTION_THRESHOLD_RATIO));
331        // Over the threshold -> compact.
332        assert!(should_compact(801, max_tokens, COMPACTION_THRESHOLD_RATIO));
333        // A ratio of 1.0 effectively disables threshold-based compaction.
334        assert!(!should_compact(1000, max_tokens, 1.0));
335        // Zero max_tokens never compacts.
336        assert!(!should_compact(100, 0, COMPACTION_THRESHOLD_RATIO));
337    }
