323    /// Append a chat message, recording its timestamp for `/timestamps`.
324    pub(crate) fn push_msg(&mut self, msg: impl Into<String>) {
325        self.message_timestamps.push(Utc::now());
326        self.messages.push(msg.into());
327    }
328
329    /// Commit any in-progress streaming response to the message log so that a
330    /// newly submitted user message appears AFTER it, preserving chat order.
331    pub(crate) fn commit_stream(&mut self) {
332        if let Some(stream) = self.current_stream.take() {
333            if !stream.is_empty() {
334                self.push_msg(stream);
335                self.stream_committed_index = Some(self.messages.len() - 1);
336            }
337        }
338    }

341/// Run the TUI event loop.
342pub async fn run_tui<B: ratatui::backend::Backend<Error = std::io::Error>>(
343    terminal: &mut Terminal<B>,
344    app: &mut App,
345) -> std::io::Result<()> {
346    loop {
347        // Drain ALL pending engine events BEFORE drawing, so the render
348        // always shows the latest state (not the state from the previous cycle)
349        loop {
350            match app.event_rx.try_recv() {
351                Ok(event) => {
352                    app.handle_event(event);
353                }
354                Err(tokio::sync::mpsc::error::TryRecvError::Disconnected) => {
355                    return Ok(());
356                }
357                Err(_) => break, // Empty, continue to draw
358            }
359        }
360
361        // Draw the UI (now with up-to-date state)
362        app.frame_count = app.frame_count.wrapping_add(1);
363        app.toasts.tick(Instant::now());
364        terminal.draw(|f| render(f, app))?;
365
366        // Check for keyboard input (with timeout for responsiveness)
367        if event::poll(std::time::Duration::from_millis(50))?
368            && let Event::Key(key) = event::read()?
369            && key.kind == KeyEventKind::Press
370        {
371            app.handle_key(key.code, key.modifiers);
372        }
373
374        if app.should_exit {
375            break;
376        }
377    }
378
379    Ok(())
380}
