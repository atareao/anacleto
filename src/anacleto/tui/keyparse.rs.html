7/// Build a `KeyEvent` for a character, optionally with the Ctrl modifier.
8pub(crate) fn key_event(ch: char, ctrl: bool) -> KeyEvent {
9    let modifiers = if ctrl {
10        KeyModifiers::CONTROL
11    } else {
12        KeyModifiers::NONE
13    };
14    KeyEvent::new(KeyCode::Char(ch), modifiers)
15}

87/// Format a single key event as a short human-readable string.
88fn format_key(key: &KeyEvent) -> String {
89    let mut s = String::new();
90    if key.modifiers.contains(KeyModifiers::CONTROL) {
91        s.push_str("Ctrl+");
92    }
93    if key.modifiers.contains(KeyModifiers::ALT) {
94        s.push_str("Alt+");
95    }
96    if key.modifiers.contains(KeyModifiers::SHIFT) {
97        s.push_str("Shift+");
98    }
99    match key.code {
100        KeyCode::Char(c) => s.push(c),
101        KeyCode::Enter => s.push_str("Enter"),
102        KeyCode::Esc => s.push_str("Esc"),
103        KeyCode::Backspace => s.push_str("Backspace"),
104        KeyCode::Tab => s.push_str("Tab"),
105        KeyCode::Left => s.push_str("Left"),
106        KeyCode::Right => s.push_str("Right"),
107        KeyCode::Up => s.push_str("Up"),
108        KeyCode::Down => s.push_str("Down"),
109        KeyCode::Home => s.push_str("Home"),
110        KeyCode::End => s.push_str("End"),
111        KeyCode::Delete => s.push_str("Delete"),
112        KeyCode::Insert => s.push_str("Insert"),
113        KeyCode::PageUp => s.push_str("PageUp"),
114        KeyCode::PageDown => s.push_str("PageDown"),
115        KeyCode::F(n) => s.push_str(&format!("F{n}")),
116        _ => s.push('?'),
117    }
118    s
119}

121/// Parse an action name (e.g. `ToggleSidebar`, `toggle_sidebar`) into an
122/// [`Action`]. Returns `None` for unknown names.

124    let normalized = name.replace(['-', ' '], "_");

135        Action::ScrollUp,
136        Action::ScrollDown,
137        Action::PageUp,
138        Action::PageDown,
139        Action::Approve,
140        Action::Deny,

142        Action::FocusSidebar,
143        Action::FocusChat,
144        Action::FocusInfo,
145        Action::FocusQueue,
146        Action::FocusMcps,
147        Action::FocusSkills,
148        Action::FocusAgents,
149        Action::ClearInput,
150        Action::OpenPromptQueue,
151        Action::QuickSlot1,
152        Action::QuickSlot2,
153        Action::QuickSlot3,
154        Action::QuickSlot4,
155        Action::QuickSlot5,
156        Action::QuickSlot6,
157        Action::QuickSlot7,
158        Action::QuickSlot8,
159        Action::QuickSlot9,
160        Action::CursorLeft,
161        Action::CursorRight,

163        Action::CursorWordRight,
164        Action::CursorHome,
165        Action::CursorEnd,
166        Action::DeleteChar,
167        Action::DeleteCharBefore,
168        Action::DeleteWordBefore,
169        Action::DeleteToStart,
170        Action::DeleteToEnd,
171        Action::HistoryUp,
172        Action::HistoryDown,
173        Action::TabComplete,
174        Action::InsertNewline,
175        Action::ChatTop,
176        Action::ChatBottom,
177        Action::ListUp,
178        Action::ListDown,
179        Action::ListTop,
180        Action::ListBottom,
181    ] {
182        if to_snake_case(&format!("{action:?}")) == lower {
183            return Some(action);
184        }
185    }
186    None
187}

189/// Convert a CamelCase string to snake_case (used for action names).
190fn to_snake_case(s: &str) -> String {
191    let mut out = String::new();
192    for (i, ch) in s.chars().enumerate() {
193        if ch.is_uppercase() {
194            if i > 0 {
195                out.push('_');
196            }
197            out.push(ch.to_ascii_lowercase());
198        } else {
199            out.push(ch);
200        }
201    }
202    out
203}

205/// Parse a key string (e.g. `ctrl+b`, `alt+e`, `f2`, `enter`, `escape`,
206/// `ctrl+shift+p`) into a [`KeyEvent`]. Returns `None` for invalid strings.
207pub(crate) fn parse_key(s: &str) -> Option<KeyEvent> {
208    let mut ctrl = false;
209    let mut alt = false;
210    let mut shift = false;
211    let mut key = s;
212    for part in s.split('+') {
213        match part.trim().to_lowercase().as_str() {
214            "ctrl" => ctrl = true,
215            "alt" => alt = true,
216            "shift" => shift = true,
217            _ => key = part,
218        }
219    }
220    let mut modifiers = KeyModifiers::NONE;
221    if ctrl {
222        modifiers.insert(KeyModifiers::CONTROL);
223    }
224    if alt {
225        modifiers.insert(KeyModifiers::ALT);
226    }
227    if shift {
228        modifiers.insert(KeyModifiers::SHIFT);
229    }
230    let code = match key.trim().to_lowercase().as_str() {
231        "enter" => KeyCode::Enter,
232        "escape" | "esc" => KeyCode::Esc,
233        "pageup" | "pgup" => KeyCode::PageUp,
234        "pagedown" | "pgdn" => KeyCode::PageDown,
235        "backspace" => KeyCode::Backspace,
236        "tab" => KeyCode::Tab,
237        "space" => KeyCode::Char(' '),
238        "left" => KeyCode::Left,
239        "right" => KeyCode::Right,
240        "up" => KeyCode::Up,
241        "down" => KeyCode::Down,
242        "home" => KeyCode::Home,
243        "end" => KeyCode::End,
244        "delete" | "del" => KeyCode::Delete,
245        "insert" | "ins" => KeyCode::Insert,
246        "f1" => KeyCode::F(1),
247        "f2" => KeyCode::F(2),
248        "f3" => KeyCode::F(3),
249        "f4" => KeyCode::F(4),
250        "f5" => KeyCode::F(5),
251        "f6" => KeyCode::F(6),
252        "f7" => KeyCode::F(7),
253        "f8" => KeyCode::F(8),
254        "f9" => KeyCode::F(9),
255        "f10" => KeyCode::F(10),
256        "f11" => KeyCode::F(11),
257        "f12" => KeyCode::F(12),
258        other => {
259            let mut chars = other.chars();
260            let c = chars.next()?;
261            if chars.next().is_some() || c.len_utf8() != 1 {
262                return None;
263            }
264            KeyCode::Char(c)
265        }
266    };
267    Some(KeyEvent::new(code, modifiers))
268}
