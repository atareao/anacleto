9use crate::tui::keyparse::format_keymap_table;

11/// A centered popup that shows the full keybinding table ("which-key").
12///
13/// It is a simple, statically-scrolled listing: any key press closes it.
14pub struct WhichKeyPopup {
15    /// Whether the popup is currently visible.
16    pub visible: bool,
17}
18
19impl WhichKeyPopup {
20    /// Create a hidden which-key popup.
21    pub fn new() -> Self {
22        Self { visible: false }
23    }
24
25    /// Render the popup centered over `area` if it is visible.
26    pub fn render(&self, f: &mut Frame, area: Rect) {
27        if !self.visible {
28            return;
29        }
30
31        let width = area.width.min(70);
32        let height = area.height.min(26);
33        let x = area.x + (area.width.saturating_sub(width)) / 2;
34        let y = area.y + (area.height.saturating_sub(height)) / 2;
35        let popup_area = Rect::new(x, y, width, height);
36
37        let overlay = Clear;
38        f.render_widget(overlay, popup_area);
39
40        let table = format_keymap_table();
41        let mut lines: Vec<Line> = table
42            .lines()
43            .map(|l| {
44                Line::from(Span::styled(
45                    l.to_string(),
46                    Style::default().fg(Color::White),
47                ))
48            })
49            .collect();
50        lines.push(Line::from(""));
51        lines.push(Line::from(Span::styled(
52            " Pulsa la tecla del atajo o Esc para cerrar ",
53            Style::default().fg(Color::Cyan).add_modifier(Modifier::DIM),
54        )));
55
56        let paragraph = Paragraph::new(lines)
57            .block(
58                Block::default()
59                    .borders(Borders::ALL)
60                    .border_type(BorderType::Rounded)
61                    .border_style(Style::default().fg(Color::Cyan))
62                    .title(" Atajos de teclado ")
63                    .style(Style::default().bg(Color::Rgb(20, 20, 30))),
64            )
65            .scroll((0, 0));
66
67        f.render_widget(paragraph, popup_area);
68    }
69}
70
71impl Default for WhichKeyPopup {
72    fn default() -> Self {
73        Self::new()
74    }
