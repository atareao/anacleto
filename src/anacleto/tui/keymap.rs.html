7/// A user-facing action that can be triggered by one or more key bindings.

20    /// Show the which-key popup.

22    /// Open the model picker.
23    OpenModelPicker,
24    /// Open the external editor.
25    OpenEditor,
26    /// Scroll the chat up by one line.
27    ScrollUp,
28    /// Scroll the chat down by one line.
29    ScrollDown,
30    /// Scroll the chat up by a page.
31    PageUp,
32    /// Scroll the chat down by a page.
33    PageDown,

39    FocusInput,

43    FocusChat,

60/// Central mapping of actions to the key events that trigger them.
61///
62/// Multiple keys can be bound to a single action. The defaults are sensible
63/// for a TUI; the map is intentionally data-driven so it can later be loaded
64/// from configuration.
65pub struct Keymap {
66    bindings: HashMap<Action, Vec<KeyEvent>>,
67}
68
69impl Keymap {
70    /// Create an empty keymap.
71    pub fn new() -> Self {
72        Self {
73            bindings: HashMap::new(),
74        }
75    }
76
77    /// Bind one or more keys to an action (replacing any previous bindings).
78    pub fn bind(&mut self, action: Action, keys: Vec<KeyEvent>) {
79        self.bindings.insert(action, keys);
80    }
81
82    /// Return the keys currently bound to an action (empty if none).
83    pub fn resolve(&self, action: Action) -> Vec<KeyEvent> {
84        self.bindings.get(&action).cloned().unwrap_or_default()
85    }
86
87    /// Whether the given key event triggers the given action.
88    pub fn matches(&self, key: KeyEvent, action: Action) -> bool {
89        self.bindings
90            .get(&action)
91            .map(|keys| keys.contains(&key))
92            .unwrap_or(false)
93    }
94
95    /// Apply user-provided keybinding overrides from configuration.
96    ///
97    /// Each entry maps an action name (e.g. `ToggleSidebar`) to a list of key
98    /// strings (e.g. `ctrl+b`, `f2`, `enter`). Invalid action names or key
99    /// strings are silently ignored; valid bindings replace the defaults.
100    pub fn apply_overrides(&mut self, overrides: &KeymapConfig) {
101        for (action_name, keys) in &overrides.bindings {
102            let Some(action) = parse_action(action_name) else {
103                continue;
104            };
105            let mut parsed = Vec::new();
106            for k in keys {
107                if let Some(ke) = parse_key(k) {
108                    parsed.push(ke);
109                }
110            }
111            if !parsed.is_empty() {
112                self.bind(action, parsed);
113            }
114        }
115    }
116}

187/// Build a `KeyEvent` for a character, optionally with the Ctrl modifier.
188pub fn key_event(ch: char, ctrl: bool) -> KeyEvent {
189    let modifiers = if ctrl {
190        KeyModifiers::CONTROL
191    } else {
192        KeyModifiers::NONE
193    };
194    KeyEvent::new(KeyCode::Char(ch), modifiers)
195}

241/// Format a single key event as a short human-readable string.
242fn format_key(key: &KeyEvent) -> String {
243    let mut s = String::new();
244    if key.modifiers.contains(KeyModifiers::CONTROL) {
245        s.push_str("Ctrl+");
246    }
247    match key.code {
248        KeyCode::Char(c) => s.push(c),
249        KeyCode::Enter => s.push_str("Enter"),
250        KeyCode::Esc => s.push_str("Esc"),
251        KeyCode::PageUp => s.push_str("PageUp"),
252        KeyCode::PageDown => s.push_str("PageDown"),
253        _ => s.push('?'),
254    }
255    s
256}

258/// Parse an action name (e.g. `ToggleSidebar`, `toggle_sidebar`) into an
259/// [`Action`]. Returns `None` for unknown names.
260fn parse_action(name: &str) -> Option<Action> {
261    let normalized = name.replace(['-', ' '], "_");
262    let lower = to_snake_case(&normalized);
263    for action in [
264        Action::Send,
265        Action::CancelInput,
266        Action::Quit,
267        Action::ToggleSidebar,
268        Action::ToggleDiffViewer,
269        Action::OpenWhichKey,
270        Action::OpenModelPicker,
271        Action::OpenEditor,
272        Action::ScrollUp,
273        Action::ScrollDown,
274        Action::PageUp,
275        Action::PageDown,
276        Action::Approve,
277        Action::Deny,
278        Action::FocusInput,
279        Action::FocusSidebar,
280        Action::FocusChat,
281        Action::ClearInput,
282        Action::OpenPromptQueue,
283        Action::QuickSlot1,
284        Action::QuickSlot2,
285        Action::QuickSlot3,
286        Action::QuickSlot4,
287        Action::QuickSlot5,
288        Action::QuickSlot6,
289        Action::QuickSlot7,
290        Action::QuickSlot8,
291        Action::QuickSlot9,
292    ] {
293        if to_snake_case(&format!("{action:?}")) == lower {
294            return Some(action);
295        }
296    }
297    None
298}

300/// Convert a CamelCase string to snake_case (used for action names).
301fn to_snake_case(s: &str) -> String {
302    let mut out = String::new();
303    for (i, ch) in s.chars().enumerate() {
304        if ch.is_uppercase() {
305            if i > 0 {
306                out.push('_');
307            }
308            out.push(ch.to_ascii_lowercase());
309        } else {
310            out.push(ch);
311        }
312    }
313    out
314}

316/// Parse a key string (e.g. `ctrl+b`, `alt+e`, `f2`, `enter`, `escape`,
317/// `ctrl+shift+p`) into a [`KeyEvent`]. Returns `None` for invalid strings.
318fn parse_key(s: &str) -> Option<KeyEvent> {
319    let mut ctrl = false;
320    let mut alt = false;
321    let mut shift = false;
322    let mut key = s;
323    for part in s.split('+') {
324        match part.trim().to_lowercase().as_str() {
325            "ctrl" => ctrl = true,
326            "alt" => alt = true,
327            "shift" => shift = true,
328            _ => key = part,
329        }
330    }
331    let mut modifiers = KeyModifiers::NONE;
332    if ctrl {
333        modifiers.insert(KeyModifiers::CONTROL);
334    }
335    if alt {
336        modifiers.insert(KeyModifiers::ALT);
337    }
338    if shift {
339        modifiers.insert(KeyModifiers::SHIFT);
340    }
341    let code = match key.trim().to_lowercase().as_str() {
342        "enter" => KeyCode::Enter,
343        "escape" | "esc" => KeyCode::Esc,
344        "pageup" | "pgup" => KeyCode::PageUp,
345        "pagedown" | "pgdn" => KeyCode::PageDown,
346        "backspace" => KeyCode::Backspace,
347        "tab" => KeyCode::Tab,
348        "space" => KeyCode::Char(' '),
349        "left" => KeyCode::Left,
350        "right" => KeyCode::Right,
351        "up" => KeyCode::Up,
352        "down" => KeyCode::Down,
353        "f1" => KeyCode::F(1),
354        "f2" => KeyCode::F(2),
355        "f3" => KeyCode::F(3),
356        "f4" => KeyCode::F(4),
357        "f5" => KeyCode::F(5),
358        "f6" => KeyCode::F(6),
359        "f7" => KeyCode::F(7),
360        "f8" => KeyCode::F(8),
361        "f9" => KeyCode::F(9),
362        "f10" => KeyCode::F(10),
363        "f11" => KeyCode::F(11),
364        "f12" => KeyCode::F(12),
365        other => {
366            let mut chars = other.chars();
367            let c = chars.next()?;
368            if chars.next().is_some() || c.len_utf8() != 1 {
369                return None;
370            }
371            KeyCode::Char(c)
372        }
373    };
374    Some(KeyEvent::new(code, modifiers))
375}
