1//! The `glob` tool: list files matching a glob pattern within the workspace.
2
3use std::path::{Path, PathBuf};
4
5use crate::llm::types::{ToolCall, ToolDefinition};
6use crate::permissions::checker::check_fs_read;
7use crate::permissions::types::Permissions;
8
9/// Maximum number of paths returned per `glob` call.
10const MAX_PATHS: usize = 1000;
11
12/// Tool definition for the `glob` tool.
13pub fn glob_tool_definition() -> ToolDefinition {
14    ToolDefinition {
15        name: "glob".to_string(),
16        description: "List files within the workspace that match a glob pattern. \
17                       Patterns are relative to the workspace root and support \
18                       '*', '?', '**' and '[a-z]' (e.g. 'src/**/*.rs'). Returns \
19                       the list of matching relative paths."
20            .to_string(),
21        input_schema: serde_json::json!({
22            "type": "object",
23            "properties": {
24                "pattern": {
25                    "type": "string",
26                    "description": "Glob pattern relative to the workspace root."
27                }
28            },
29            "required": ["pattern"]
30        }),
31    }
32}
33
34/// Recursively collect all files under `dir`.
35fn walk_files(dir: &Path, out: &mut Vec<PathBuf>) {
36    if let Ok(entries) = std::fs::read_dir(dir) {
37        for entry in entries.flatten() {
38            let path = entry.path();
39            if path.is_dir() {
40                walk_files(&path, out);
41            } else if path.is_file() {
42                out.push(path);
43            }
44        }
45    }
46}

48/// Execute a `glob` tool call.
49pub async fn execute_glob_tool(
50    workspace: &Path,
51    permissions: &Permissions,
52    tool_call: &ToolCall,
53) -> Result<String, String> {
54    let args: serde_json::Value = serde_json::from_str(&tool_call.function.arguments)
55        .map_err(|e| format!("Failed to parse glob arguments: {e}"))?;
56    let pattern = args
57        .get("pattern")
58        .and_then(|v| v.as_str())
59        .ok_or_else(|| "glob requires 'pattern'".to_string())?;
60
61    check_fs_read(permissions).map_err(|e| format!("Permission denied: {e}"))?;
62
63    let workspace_canon = workspace
64        .canonicalize()
65        .map_err(|e| format!("Invalid workspace '{}': {e}", workspace.display()))?;
66
67    let mut files: Vec<PathBuf> = Vec::new();
68    walk_files(&workspace_canon, &mut files);
69
70    let mut matches: Vec<String> = Vec::new();
71    for file in files {
72        let rel = file
73            .strip_prefix(&workspace_canon)
74            .unwrap_or(&file)
75            .to_string_lossy()
76            .into_owned();
77        if crate::tools::pattern::glob_match(pattern, &rel) {
78            matches.push(rel);
79        }
80    }
81
82    matches.sort();
83    if matches.is_empty() {
84        return Ok(format!("No files match pattern: {pattern}"));
85    }
86
87    let mut out = String::new();
88    for (i, m) in matches.iter().enumerate() {
89        if i >= MAX_PATHS {
90            out.push_str(&format!(
91                "... (truncated at {MAX_PATHS} paths; refine your pattern)\n"
92            ));
93            break;
94        }
95        out.push_str(m);
96        out.push('\n');
97    }
98    Ok(out)
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use crate::config::PermissionConfig;
105    use crate::llm::types::ToolFunction;
106    use crate::permissions::types::Permissions;
107
108    fn temp_workspace() -> PathBuf {
109        let dir = std::env::temp_dir().join(format!("anacleto_glob_test_{}", uuid::Uuid::new_v4()));
110        std::fs::create_dir_all(&dir).unwrap();
111        dir
112    }
113
114    fn tool_call(args: &str) -> ToolCall {
115        ToolCall {
116            id: "call_glob".into(),
117            call_type: "function".into(),
118            function: ToolFunction {
119                name: "glob".into(),
120                arguments: args.into(),
121            },
122        }
123    }
124
125    fn allow_all() -> Permissions {
126        Permissions::from_config(&PermissionConfig {
127            deny: vec![],
128            allow: vec![],
129        })
130    }
131
132    #[tokio::test]
133    async fn glob_lists_matching_files() {
134        let ws = temp_workspace();
135        std::fs::create_dir_all(ws.join("src/sub")).unwrap();
136        std::fs::write(ws.join("src/main.rs"), "").unwrap();
137        std::fs::write(ws.join("src/sub/lib.rs"), "").unwrap();
138        std::fs::write(ws.join("README.md"), "").unwrap();
139
140        let result = execute_glob_tool(&ws, &allow_all(), &tool_call(r#"{"pattern":"**/*.rs"}"#))
141            .await
142            .unwrap();
143        assert!(result.contains("src/main.rs"));
144        assert!(result.contains("src/sub/lib.rs"));
145        assert!(!result.contains("README.md"));
146
147        std::fs::remove_dir_all(&ws).unwrap();
148    }
149
150    #[tokio::test]
151    async fn glob_no_matches() {
152        let ws = temp_workspace();
153        std::fs::write(ws.join("a.txt"), "").unwrap();
154        let result = execute_glob_tool(&ws, &allow_all(), &tool_call(r#"{"pattern":"*.py"}"#))
155            .await
156            .unwrap();
157        assert!(result.contains("No files match"));
158        std::fs::remove_dir_all(&ws).unwrap();
159    }
