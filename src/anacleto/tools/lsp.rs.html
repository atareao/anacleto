1//! The `lsp_query` tool: query a Language Server Protocol (LSP) server for
2//! hover, definition, references or diagnostics.
3//!
4//! This tool launches a language server process (e.g. `rust-analyzer`) over
5//! stdio, so it requires the `command.run` permission.
6
7use crate::llm::types::{ToolCall, ToolDefinition};
8use crate::lsp::{LspClient, LspPosition, LspQueryType, default_server_for_extension, path_to_uri};
9use crate::permissions::checker::check_command_run;
10use crate::permissions::types::Permissions;
11
12/// Tool definition for the `lsp_query` tool.
13pub fn lsp_query_tool_definition() -> ToolDefinition {
14    ToolDefinition {
15        name: "lsp_query".to_string(),
16        description: "Query a Language Server Protocol (LSP) server for code \
17                       intelligence. Provide `server_command` (e.g. 'rust-analyzer' \
18                       or 'typescript-language-server'), the `file_path` to analyze, \
19                       the 0-based `line` and `character` position, and a `query_type` \
20                       of 'hover', 'definition', 'references' or 'diagnostic'. \
21                       If `server_command` is omitted, a default is chosen from the \
22                       file extension. Launches the server process, so it requires \
23                       the command.run permission."
24            .to_string(),
25        input_schema: serde_json::json!({
26            "type": "object",
27            "properties": {
28                "server_command": {
29                    "type": "string",
30                    "description": "The LSP server executable (e.g. 'rust-analyzer'). Optional; inferred from the file extension if omitted."
31                },
32                "file_path": {
33                    "type": "string",
34                    "description": "Absolute path to the file to analyze."
35                },
36                "line": {
37                    "type": "integer",
38                    "minimum": 0,
39                    "description": "0-based line number of the position (ignored for 'diagnostic')."
40                },
41                "character": {
42                    "type": "integer",
43                    "minimum": 0,
44                    "description": "0-based character offset of the position (ignored for 'diagnostic')."
45                },
46                "query_type": {
47                    "type": "string",
48                    "enum": ["hover", "definition", "references", "diagnostic"],
49                    "description": "The kind of LSP query to perform."
50                }
51            },
52            "required": ["file_path", "query_type"]
53        }),
54    }
55}

57/// Execute an `lsp_query` tool call.
58pub async fn execute_lsp_query_tool(
59    permissions: &Permissions,
60    tool_call: &ToolCall,
61) -> Result<String, String> {
62    check_command_run(permissions).map_err(|e| format!("Permission denied: {e}"))?;
63
64    let args: serde_json::Value = serde_json::from_str(&tool_call.function.arguments)
65        .map_err(|e| format!("Failed to parse lsp_query arguments: {e}"))?;
66
67    let file_path = args
68        .get("file_path")
69        .and_then(|v| v.as_str())
70        .ok_or_else(|| "lsp_query requires 'file_path'".to_string())?;
71    let query_type = args
72        .get("query_type")
73        .and_then(|v| v.as_str())
74        .ok_or_else(|| "lsp_query requires 'query_type'".to_string())?;
75    let query_type = LspQueryType::parse(query_type)
76        .ok_or_else(|| format!("Unknown query_type '{query_type}'"))?;
77
78    let line = args.get("line").and_then(|v| v.as_u64()).unwrap_or(0) as u32;
79    let character = args.get("character").and_then(|v| v.as_u64()).unwrap_or(0) as u32;
80
81    // Resolve the server command: explicit, or inferred from the extension.
82    let server_command = args
83        .get("server_command")
84        .and_then(|v| v.as_str())
85        .map(|s| s.to_string())
86        .or_else(|| {
87            let ext = std::path::Path::new(file_path)
88                .extension()
89                .and_then(|e| e.to_str())
90                .unwrap_or("");
91            default_server_for_extension(ext).map(|s| s.to_string())
92        })
93        .ok_or_else(|| {
94            "No server_command provided and no default LSP server is known for this file extension"
95                .to_string()
96        })?;
97
98    let uri = path_to_uri(file_path);
99    let root_uri = path_to_uri(
100        std::path::Path::new(file_path)
101            .parent()
102            .and_then(|p| p.to_str())
103            .unwrap_or("."),
104    );
105
106    let position = LspPosition { line, character };
107    let result = LspClient::query_once(&server_command, &[], &root_uri, &uri, position, query_type)
108        .await
109        .map_err(|e| format!("LSP query failed: {e}"))?;
110
111    Ok(result)
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117    use crate::config::PermissionConfig;
118    use crate::llm::types::ToolFunction;
119    use crate::permissions::types::Permissions;
120
121    fn tool_call(args: &str) -> ToolCall {
122        ToolCall {
123            id: "call_lsp".into(),
124            call_type: "function".into(),
125            function: ToolFunction {
126                name: "lsp_query".into(),
127                arguments: args.into(),
128            },
129        }
130    }
131
132    fn deny_command() -> Permissions {
133        Permissions::from_config(&PermissionConfig {
134            deny: vec!["command.run".into()],
135            allow: vec![],
136        })
137    }
138
139    fn allow_all() -> Permissions {
140        Permissions::from_config(&PermissionConfig {
141            deny: vec![],
142            allow: vec![],
143        })
144    }
145
146    #[tokio::test]
147    async fn lsp_query_denied_without_command_run() {
148        let result = execute_lsp_query_tool(
149            &deny_command(),
150            &tool_call(r#"{"file_path":"/tmp/a.rs","query_type":"hover"}"#),
151        )
152        .await;
153        assert!(result.is_err());
154        assert!(result.unwrap_err().contains("Permission denied"));
155    }
156
157    #[tokio::test]
158    async fn lsp_query_missing_file_path_errors() {
159        let result =
160            execute_lsp_query_tool(&allow_all(), &tool_call(r#"{"query_type":"hover"}"#)).await;
161        assert!(result.is_err());
162        assert!(result.unwrap_err().contains("requires 'file_path'"));
163    }
164
165    #[tokio::test]
166    async fn lsp_query_unknown_query_type_errors() {
167        let result = execute_lsp_query_tool(
168            &allow_all(),
169            &tool_call(r#"{"file_path":"/tmp/a.rs","query_type":"bogus"}"#),
170        )
171        .await;
172        assert!(result.is_err());
173        assert!(result.unwrap_err().contains("Unknown query_type"));
174    }
175
176    #[tokio::test]
177    async fn lsp_query_unknown_extension_and_no_server_errors() {
178        // No server_command and an unknown extension -> clear error, no process spawned.
179        let result = execute_lsp_query_tool(
180            &allow_all(),
181            &tool_call(r#"{"file_path":"/tmp/a.xyz","query_type":"hover"}"#),
182        )
183        .await;
184        assert!(result.is_err());
185        assert!(result.unwrap_err().contains("No server_command provided"));
186    }
187}
