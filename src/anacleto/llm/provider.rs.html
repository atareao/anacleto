24/// Trait for LLM provider implementations.
25#[async_trait]
26pub trait LlmProvider: Send + Sync {
27    /// Send a request and get a complete response.
28    async fn complete(&self, request: LlmRequest) -> Result<LlmResponse>;

30    /// Send a request and get a streaming response.
31    async fn complete_stream(
32        &self,
33        request: LlmRequest,

35
36    /// Return the context window (max tokens) for this provider.
37    fn context_window(&self) -> usize;
38
39    /// Fetch the model's context window from the provider API.
40    async fn fetch_context_window(&self) -> Result<usize>;
41
42    /// Override the context window (used after fetching from the API).
43    fn set_context_window(&self, value: usize);
44
45    /// Input price in USD per million tokens.
46    fn input_price_per_million(&self) -> f64;
47
48    /// Output price in USD per million tokens.

52/// Factory for creating LLM providers.
53pub fn create_provider(config: &LlmProviderConfig) -> Box<dyn LlmProvider> {
54    match config.provider_type {
55        LlmProviderType::Anthropic => Box::new(AnthropicProvider::new(config)),
56        LlmProviderType::OpenAI => Box::new(OpenAIProvider::new(config)),
57        LlmProviderType::OpenRouter => Box::new(OpenRouterProvider::new(config)),
58        LlmProviderType::Ollama => Box::new(OllamaProvider::new(config)),
59        LlmProviderType::Bedrock => Box::new(BedrockProvider::new(config)),
60        LlmProviderType::Azure => Box::new(AzureProvider::new(config)),
61        LlmProviderType::Google => Box::new(GoogleProvider::new(config)),
62    }
63}

77/// Convert a `Vec<LlmMessage>` into OpenAI-compatible message list.
78pub(crate) fn into_openai_messages(messages: Vec<LlmMessage>) -> Vec<OpenAiMessage> {
79    messages
80        .into_iter()
81        .map(|m| OpenAiMessage {
82            role: match m.role {
83                MessageRole::User => "user".into(),
84                MessageRole::Assistant => "assistant".into(),
85                MessageRole::System => "system".into(),
86                MessageRole::Tool => "tool".into(),
87            },
88            content: m.content,
89            tool_calls: m.tool_calls.map(|calls| {
90                calls
91                    .into_iter()
92                    .map(|tc| OpenAiToolCall {
93                        id: tc.id,
94                        type_: tc.call_type,
95                        function: OpenAiFunctionCall {
96                            name: tc.function.name,
97                            arguments: tc.function.arguments,
98                        },
99                    })
100                    .collect()
101            }),
102            tool_call_id: m.tool_call_id,
103        })
104        .collect()
105}

107/// Convert a `Vec<LlmMessage>` into Anthropic-compatible message list,
108/// extracting the system message separately.
109///
110/// Prompt caching for Anthropic is handled via the top-level `cache_control`
111/// field on the request (automatic caching), which is set in
112/// [`AnthropicProvider::complete`]. Message-level `cache_control` breakpoints
113/// are intentionally NOT injected here: they are only valid inside a content
114/// block array, and the automatic top-level caching already covers the
115/// prompt-prefix caching use case.
116pub(crate) fn into_anthropic_messages(
117    messages: Vec<LlmMessage>,
118) -> (Option<String>, Vec<AnthropicMessage>) {
119    let mut system = None;
120    let mut msgs = Vec::new();
121    for m in messages {
122        match m.role {
123            MessageRole::System => system = Some(m.content),
124            _ => msgs.push(AnthropicMessage {
125                role: match m.role {
126                    MessageRole::User => "user".into(),
127                    MessageRole::Assistant => "assistant".into(),
128                    MessageRole::Tool => "user".into(),
129                    _ => "user".into(),
130                },
131                content: m.content,
132            }),
133        }
134    }
135    (system, msgs)
136}

182/// Parse SSE data lines from a byte stream, yielding parsed `OpenAiStreamChunk`s.
183/// Returns `None` when the stream is done (e.g. `[DONE]` received).
184pub(crate) fn parse_sse_line(line: &str) -> Option<OpenAiStreamChunk> {
185    let line = line.trim();
186    if line.is_empty() || line.starts_with(':') {
187        return None;
188    }
189    if line == "data: [DONE]" {
190        // Signal end-of-stream
191        return None;
192    }
193    if let Some(data) = line.strip_prefix("data: ") {
194        serde_json::from_str::<OpenAiStreamChunk>(data)
195            .map_err(|e| tracing::warn!("Failed to parse SSE chunk: {e}"))
196            .ok()
197    } else {
198        None
199    }
200}

202/// Strip the `~` routing prefix from a model name if present.
203/// The `~` prefix is used internally by Anacleto to hint at provider routing
204/// (e.g. `~deepseek/deepseek-v4-flash` → `deepseek/deepseek-v4-flash`).
205pub(crate) fn strip_model_prefix(model: &str) -> String {
206    model.strip_prefix('~').unwrap_or(model).to_string()
207}

263    async fn fetch_context_window(&self) -> Result<usize> {
264        let model = strip_model_prefix(&self.config.model);
265        // OpenRouter's single-model endpoint `GET /models/{model}` returns 404
266        // for model ids that contain a `/` (e.g. "deepseek/deepseek-v4-flash"),
267        // so fetch the full list and look the model up by id instead.
268        let url = format!("{}/models", self.base_url());
269        let resp = self
270            .client
271            .get(&url)
272            .header("Authorization", format!("Bearer {}", self.api_key()?))
273            .send()
274            .await
275            .map_err(|e| Error::Provider(format!("OpenRouter model info request failed: {e}")))?;
276
277        if !resp.status().is_success() {
278            let status = resp.status();
279            let text = resp.text().await.unwrap_or_default();
280            return Err(Error::Provider(format!(
281                "OpenRouter model info returned {status}: {text}"
282            )));
283        }
284
285        let list: OpenRouterModelList = resp
286            .json()
287            .await
288            .map_err(|e| Error::Provider(format!("OpenRouter model info parse failed: {e}")))?;
289
290        list.data
291            .into_iter()
292            .find(|m| m.id == model)
293            .and_then(|m| m.context_length)
294            .ok_or_else(|| {
295                Error::Provider(format!(
296                    "OpenRouter model '{model}' not found or has no context_length"
297                ))
298            })
299    }

301    async fn complete(&self, request: LlmRequest) -> Result<LlmResponse> {
302        let model = strip_model_prefix(&request.model);
303        let body = OpenAiChatRequest {
304            model: model.clone(),
305            messages: into_openai_messages(request.messages),
306            max_tokens: request.max_tokens,
307            temperature: request.temperature,
308            stream: false,
309            tools: if request.tools.is_empty() {
310                None
311            } else {
312                Some(into_openai_tools(request.tools.clone()))
313            },
314            tool_choice: if request.tools.is_empty() {
315                None
316            } else {
317                Some(serde_json::json!("auto"))
318            },
319            stream_options: Some(serde_json::json!({"include_usage": true})),
320        };
321
322        let url = format!("{}/chat/completions", self.base_url());
323        let resp = self
324            .client
325            .post(&url)
326            .header("Authorization", format!("Bearer {}", self.api_key()?))
327            // OpenRouter-specific headers for app ranking
328            .header("HTTP-Referer", "https://github.com/anacleto")
329            .header("X-Title", "Anacleto")
330            .json(&body)
331            .send()
332            .await
333            .map_err(|e| Error::Provider(format!("OpenRouter request failed: {e}")))?;
334
335        if !resp.status().is_success() {
336            let status = resp.status();
337            let text = resp.text().await.unwrap_or_default();
338            return Err(Error::Provider(format!(
339                "OpenRouter returned {status}: {text}"
340            )));
341        }
342
343        let data: OpenAiChatResponse = resp
344            .json()
345            .await
346            .map_err(|e| Error::Provider(format!("OpenRouter parse failed: {e}")))?;
347
348        let choice = data
349            .choices
350            .into_iter()
351            .next()
352            .ok_or_else(|| Error::Provider("OpenRouter returned no choices".into()))?;
353
354        Ok(LlmResponse {
355            content: choice.message.content.unwrap_or_default(),
356            tool_calls: choice
357                .message
358                .tool_calls
359                .unwrap_or_default()
360                .into_iter()
361                .map(|tc| ToolCall {
362                    id: tc.id,
363                    call_type: tc.type_,
364                    function: ToolFunction {
365                        name: tc.function.name,
366                        arguments: tc.function.arguments,
367                    },
368                })
369                .collect(),
370            finish_reason: choice.finish_reason,
371            usage: data.usage.map(|u| LlmUsage {
372                prompt_tokens: u.prompt_tokens,
373                completion_tokens: u.completion_tokens,
374                total_tokens: u.total_tokens,
375            }),
376            thinking: None,
377        })
378    }

380    async fn complete_stream(
381        &self,
382        request: LlmRequest,
383    ) -> Result<tokio::sync::mpsc::Receiver<Result<LlmStreamChunk>>> {
384        let model = strip_model_prefix(&request.model);
385        let body = OpenAiChatRequest {
386            model: model.clone(),
387            messages: into_openai_messages(request.messages),
388            max_tokens: request.max_tokens,
389            temperature: request.temperature,
390            stream: true,
391            tools: if request.tools.is_empty() {
392                None
393            } else {
394                Some(into_openai_tools(request.tools.clone()))
395            },
396            tool_choice: if request.tools.is_empty() {
397                None
398            } else {
399                Some(serde_json::json!("auto"))
400            },
401            stream_options: Some(serde_json::json!({"include_usage": true})),
402        };
403
404        let url = format!("{}/chat/completions", self.base_url());
405        let client = self.client.clone();
406        let api_key = self.api_key()?.to_string();
407        let (tx, rx) = tokio::sync::mpsc::channel(64);
408
409        tokio::spawn(async move {
410            let resp = match client
411                .post(&url)
412                .header("Authorization", format!("Bearer {api_key}"))
413                .header("HTTP-Referer", "https://github.com/anacleto")
414                .header("X-Title", "Anacleto")
415                .json(&body)
416                .send()
417                .await
418            {
419                Ok(r) => r,
420                Err(e) => {
421                    let _ = tx
422                        .send(Err(Error::Provider(format!(
423                            "OpenRouter stream request failed: {e}"
424                        ))))
425                        .await;
426                    return;
427                }
428            };
429
430            if !resp.status().is_success() {
431                let status = resp.status();
432                let text = resp.text().await.unwrap_or_default();
433                let _ = tx
434                    .send(Err(Error::Provider(format!(
435                        "OpenRouter stream returned {status}: {text}"
436                    ))))
437                    .await;
438                return;
439            }
440
441            let mut stream = resp.bytes_stream();
442            use futures::StreamExt;
443            let mut buf = String::new();
444            // Accumulator for streaming tool calls (OpenRouter sends partial deltas across SSE events)
445            let mut tool_calls_acc: Vec<OpenAiToolCall> = Vec::new();
446            // Tracks whether a Done chunk has already been emitted, so we never
447            // send more than one (OpenAI/OpenRouter send usage in a SEPARATE SSE
448            // event with choices=[] after the final choice).
449            let mut done_sent = false;
450
451            while let Some(chunk) = stream.next().await {
452                match chunk {
453                    Ok(bytes) => {
454                        buf.push_str(&String::from_utf8_lossy(&bytes));
455                        while let Some(line_end) = buf.find('\n') {
456                            let line = buf[..line_end].to_string();
457                            buf = buf[line_end + 1..].to_string();
458                            if let Some(chunk) = parse_sse_line(&line) {
459                                let usage = chunk.usage.clone();
460                                for choice in chunk.choices {
461                                    if let Some(content) = choice.delta.content
462                                        && !content.is_empty()
463                                    {
464                                        let _ = tx.send(Ok(LlmStreamChunk::Content(content))).await;
465                                    }
466
467                                    // Merge streaming tool call deltas by index
468                                    if let Some(tcs) = choice.delta.tool_calls {
469                                        for tc in tcs {
470                                            let idx = tc.index.unwrap_or(0);
471                                            while tool_calls_acc.len() <= idx {
472                                                tool_calls_acc.push(OpenAiToolCall {
473                                                    id: String::new(),
474                                                    type_: "function".into(),
475                                                    function: OpenAiFunctionCall {
476                                                        name: String::new(),
477                                                        arguments: String::new(),
478                                                    },
479                                                });
480                                            }
481                                            let entry = &mut tool_calls_acc[idx];
482                                            if let Some(id) = tc.id {
483                                                entry.id = id;
484                                            }
485                                            if let Some(t) = tc.type_ {
486                                                entry.type_ = t;
487                                            }
488                                            if let Some(f) = tc.function {
489                                                if let Some(name) = f.name {
490                                                    entry.function.name = name;
491                                                }
492                                                if let Some(args) = f.arguments {
493                                                    entry.function.arguments.push_str(&args);
494                                                }
495                                            }
496                                        }
497                                    }
498
499                                    if let Some(finish) = choice.finish_reason
500                                        && finish != "null"
501                                    {
502                                        // Emit completed tool calls before Done. We do NOT
503                                        // emit Done here: OpenAI/OpenRouter send usage in a
504                                        // SEPARATE SSE event (choices=[]) after this chunk,
505                                        // so emitting Done here would report 0 tokens.
506                                        if finish == "tool_calls" && !tool_calls_acc.is_empty() {
507                                            for tc in tool_calls_acc.drain(..) {
508                                                let _ = tx
509                                                    .send(Ok(LlmStreamChunk::ToolCall(ToolCall {
510                                                        id: tc.id,
511                                                        call_type: tc.type_,
512                                                        function: ToolFunction {
513                                                            name: tc.function.name,
514                                                            arguments: tc.function.arguments,
515                                                        },
516                                                    })))
517                                                    .await;
518                                            }
519                                        }
520                                    }
521                                }
522                                // OpenRouter sends usage in a SEPARATE SSE event
523                                // (choices=[]). Emit Done here with the real usage,
524                                // but only once.
525                                if let Some(u) = usage.as_ref().filter(|_| !done_sent) {
526                                    done_sent = true;
527                                    let _ = tx
528                                        .send(Ok(LlmStreamChunk::Done(LlmUsage {
529                                            prompt_tokens: u.prompt_tokens,
530                                            completion_tokens: u.completion_tokens,
531                                            total_tokens: u.total_tokens,
532                                        })))
533                                        .await;
534                                }
535                            }
536                        }
537                    }
538                    Err(e) => {
539                        let _ = tx
540                            .send(Err(Error::Provider(format!(
541                                "OpenRouter stream read error: {e}"
542                            ))))
543                            .await;
544                        return;
545                    }
546                }
547            }
548
549            // Fallback: if the provider never sent usage, emit a Done with 0
550            // tokens so the stream never hangs waiting for a terminal chunk.
551            if !done_sent {
552                let _ = tx
553                    .send(Ok(LlmStreamChunk::Done(LlmUsage {
554                        prompt_tokens: 0,
555                        completion_tokens: 0,
556                        total_tokens: 0,
557                    })))
558                    .await;
559            }
560        });
561
562        Ok(rx)
563    }
564}

566/// Registry of LLM providers, stored as Arc for sharing across agents.
567pub struct LlmProviderRegistry {
568    providers: HashMap<String, Arc<dyn LlmProvider>>,
569}
570
571impl Clone for LlmProviderRegistry {
572    fn clone(&self) -> Self {
573        Self {
574            providers: self.providers.clone(),
575        }
576    }
577}
578
579impl LlmProviderRegistry {
580    /// Creates an empty provider registry.
581    pub fn new() -> Self {
582        Self {
583            providers: HashMap::new(),
584        }
585    }
586
587    /// Registers a named LLM provider in the registry.
588    pub fn register(&mut self, name: String, provider: Arc<dyn LlmProvider>) {
589        self.providers.insert(name, provider);
590    }
591
592    /// Returns a reference to the provider registered under the given name, if any.
593    pub fn get(&self, name: &str) -> Option<Arc<dyn LlmProvider>> {
594        self.providers.get(name).cloned()
595    }
596}
597
598impl Default for LlmProviderRegistry {
599    fn default() -> Self {
600        Self::new()
601    }
