98    async fn fetch_context_window(&self) -> Result<usize> {
99        let model = strip_model_prefix(&self.config.model);
100        let body = serde_json::json!({ "model": model });
101        let resp = self
102            .client
103            .post(format!("{}/api/show", self.base_url()))
104            .json(&body)
105            .send()
106            .await
107            .map_err(|e| Error::Provider(format!("Ollama show request failed: {e}")))?;
108
109        if !resp.status().is_success() {
110            let status = resp.status();
111            let text = resp.text().await.unwrap_or_default();
112            return Err(Error::Provider(format!(
113                "Ollama show returned {status}: {text}"
114            )));
115        }
116
117        let data: OllamaShowResponse = resp
118            .json()
119            .await
120            .map_err(|e| Error::Provider(format!("Ollama show parse failed: {e}")))?;
121
122        data.model_info
123            .iter()
124            .find(|(key, _)| key.contains("context_length"))
125            .and_then(|(_, value)| value.as_u64().map(|v| v as usize))
126            .ok_or_else(|| Error::Provider("Ollama model info has no context_length".into()))
127    }

129    async fn complete(&self, request: LlmRequest) -> Result<LlmResponse> {
130        let model = strip_model_prefix(&request.model);
131        let body = OllamaChatRequest {
132            model: model.clone(),
133            messages: into_ollama_messages(request.messages),
134            stream: false,
135            options: request.temperature.map(|t| OllamaOptions {
136                temperature: Some(t),
137            }),
138        };
139
140        let resp = self
141            .client
142            .post(format!("{}/api/chat", self.base_url()))
143            .json(&body)
144            .send()
145            .await
146            .map_err(|e| Error::Provider(format!("Ollama request failed: {e}")))?;
147
148        if !resp.status().is_success() {
149            let status = resp.status();
150            let text = resp.text().await.unwrap_or_default();
151            return Err(Error::Provider(format!("Ollama returned {status}: {text}")));
152        }
153
154        let data: OllamaChatResponse = resp
155            .json()
156            .await
157            .map_err(|e| Error::Provider(format!("Ollama parse failed: {e}")))?;
158
159        Ok(LlmResponse {
160            content: data.message.content,
161            tool_calls: data
162                .message
163                .tool_calls
164                .unwrap_or_default()
165                .into_iter()
166                .filter_map(|v| {
167                    let name = v.get("function")?.get("name")?.as_str()?.to_string();
168                    let arguments = v
169                        .get("function")?
170                        .get("arguments")
171                        .map(|a| a.to_string())
172                        .unwrap_or_default();
173                    Some(ToolCall {
174                        id: format!("ollama_{}", name),
175                        call_type: "function".into(),
176                        function: ToolFunction { name, arguments },
177                    })
178                })
179                .collect(),
180            finish_reason: if data.done { "stop" } else { "unknown" }.into(),
181            usage: Some(LlmUsage {
182                prompt_tokens: data.prompt_eval_count.unwrap_or(0),
183                completion_tokens: data.eval_count.unwrap_or(0),
184                total_tokens: data.prompt_eval_count.unwrap_or(0) + data.eval_count.unwrap_or(0),
185            }),
186            thinking: None,
187        })
188    }

190    async fn complete_stream(
191        &self,
192        request: LlmRequest,
193    ) -> Result<tokio::sync::mpsc::Receiver<Result<LlmStreamChunk>>> {
194        let model = strip_model_prefix(&request.model);
195        let body = OllamaChatRequest {
196            model: model.clone(),
197            messages: into_ollama_messages(request.messages),
198            stream: true,
199            options: request.temperature.map(|t| OllamaOptions {
200                temperature: Some(t),
201            }),
202        };
203
204        let client = self.client.clone();
205        let url = format!("{}/api/chat", self.base_url());
206        let (tx, rx) = tokio::sync::mpsc::channel(64);
207
208        tokio::spawn(async move {
209            let resp = match client.post(&url).json(&body).send().await {
210                Ok(r) => r,
211                Err(e) => {
212                    let _ = tx
213                        .send(Err(Error::Provider(format!(
214                            "Ollama stream request failed: {e}"
215                        ))))
216                        .await;
217                    return;
218                }
219            };
220
221            if !resp.status().is_success() {
222                let status = resp.status();
223                let text = resp.text().await.unwrap_or_default();
224                let _ = tx
225                    .send(Err(Error::Provider(format!(
226                        "Ollama stream returned {status}: {text}"
227                    ))))
228                    .await;
229                return;
230            }
231
232            use futures::StreamExt;
233            let mut stream = resp.bytes_stream();
234            let mut buf = String::new();
235
236            while let Some(chunk) = stream.next().await {
237                match chunk {
238                    Ok(bytes) => {
239                        buf.push_str(&String::from_utf8_lossy(&bytes));
240                        while let Some(line_end) = buf.find('\n') {
241                            let line = buf[..line_end].trim().to_string();
242                            buf = buf[line_end + 1..].to_string();
243                            if line.is_empty() {
244                                continue;
245                            }
246                            if let Ok(data) = serde_json::from_str::<OllamaChatResponse>(&line) {
247                                if !data.message.content.is_empty() {
248                                    let _ = tx
249                                        .send(Ok(LlmStreamChunk::Content(data.message.content)))
250                                        .await;
251                                }
252                                if data.done {
253                                    let _ = tx
254                                        .send(Ok(LlmStreamChunk::Done(LlmUsage {
255                                            prompt_tokens: data.prompt_eval_count.unwrap_or(0),
256                                            completion_tokens: data.eval_count.unwrap_or(0),
257                                            total_tokens: data.prompt_eval_count.unwrap_or(0)
258                                                + data.eval_count.unwrap_or(0),
259                                        })))
260                                        .await;
261                                }
262                            }
263                        }
264                    }
265                    Err(e) => {
266                        let _ = tx
267                            .send(Err(Error::Provider(format!(
268                                "Ollama stream read error: {e}"
269                            ))))
270                            .await;
271                        return;
272                    }
273                }
274            }
275        });
276
277        Ok(rx)
278    }
