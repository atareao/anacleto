1//! Azure OpenAI provider.
2//!
3//! Azure OpenAI exposes an OpenAI-compatible chat completions endpoint. The
4//! `base_url` should point at the deployment endpoint
5//! (`https://<resource>.openai.azure.com/openai/deployments/<deployment>`)
6//! and the `api_key` is the Azure resource key.
7
8use async_trait::async_trait;
9
10use crate::error::Result;
11use crate::llm::provider::{LlmProvider, OpenAIProvider};
12use crate::llm::types::{LlmProviderConfig, LlmRequest, LlmResponse, LlmStreamChunk};
13
14/// Azure OpenAI LLM provider (OpenAI-compatible).
15pub struct AzureProvider {
16    inner: OpenAIProvider,
17}
18
19impl AzureProvider {
20    /// Creates a new Azure provider from the given configuration.
21    pub fn new(config: &LlmProviderConfig) -> Self {
22        let mut cfg = config.clone();
23        if cfg.base_url.is_none() {
24            cfg.base_url = Some(
25                "https://YOUR_RESOURCE.openai.azure.com/openai/deployments/YOUR_DEPLOYMENT"
26                    .to_string(),
27            );
28        }
29        Self {
30            inner: OpenAIProvider::new(&cfg),
31        }
32    }
33}
34
35#[async_trait]
36impl LlmProvider for AzureProvider {
37    async fn complete(&self, request: LlmRequest) -> Result<LlmResponse> {
38        self.inner.complete(request).await
39    }
40
41    async fn complete_stream(
42        &self,
43        request: LlmRequest,
44    ) -> Result<tokio::sync::mpsc::Receiver<Result<LlmStreamChunk>>> {
45        self.inner.complete_stream(request).await
46    }
47
48    fn context_window(&self) -> usize {
49        self.inner.context_window()
50    }
51
52    async fn fetch_context_window(&self) -> Result<usize> {
53        self.inner.fetch_context_window().await
54    }
55
56    fn set_context_window(&self, value: usize) {
57        self.inner.set_context_window(value);
58    }
59
60    fn input_price_per_million(&self) -> f64 {
61        self.inner.input_price_per_million()
62    }
63
64    fn output_price_per_million(&self) -> f64 {
65        self.inner.output_price_per_million()
66    }
67}

69#[cfg(test)]
70mod tests {
71    use super::*;
72    use crate::llm::types::CacheControl;
73
74    #[test]
75    fn test_azure_provider_construction() {
76        let config = LlmProviderConfig {
77            provider_type: crate::llm::types::LlmProviderType::Azure,
78            api_key: Some("azure-key".into()),
79            model: "gpt-4o".into(),
80            base_url: Some("https://my-resource.openai.azure.com/openai/deployments/gpt-4o".into()),
81            context_window: 128_000,
82            input_price_per_million: 3.0,
83            output_price_per_million: 15.0,
84            cache_control: CacheControl::Auto,
85            thinking_budget_tokens: None,
86        };
87        let provider = AzureProvider::new(&config);
88        assert_eq!(provider.context_window(), 128_000);
89        assert_eq!(provider.output_price_per_million(), 15.0);
90    }
91
92    #[test]
93    fn test_azure_default_base_url() {
94        let config = LlmProviderConfig {
95            provider_type: crate::llm::types::LlmProviderType::Azure,
96            api_key: Some("azure-key".into()),
97            model: "gpt-4o".into(),
98            base_url: None,
99            context_window: 128_000,
100            input_price_per_million: 3.0,
101            output_price_per_million: 15.0,
102            cache_control: CacheControl::Auto,
103            thinking_budget_tokens: None,
104        };
105        let provider = AzureProvider::new(&config);
106        assert!(provider.inner.base_url().contains("openai.azure.com"));
107    }
