6/// Load and merge configuration from global and project paths.
7///
8/// `explicit_config` is an optional path to a project config file (from the
9/// `--config` CLI flag). When provided, it overrides auto-detection and its
10/// parent directory becomes the project root for agent discovery.
11pub fn load_config(explicit_config: Option<&Path>) -> Result<Config> {
12    let global_path = global_config_path();
13
14    // Determine the project root. If an explicit config was given, its parent
15    // directory is the project root; otherwise walk up from the CWD.
16    let explicit_root = explicit_config.and_then(|p| p.parent());
17    let project_path = match explicit_config {
18        Some(p) => p.to_path_buf(),
19        None => crate::config::paths::project_config_path(None),
20    };
21
22    let mut config = Config::default();
23
24    // Load global config first
25    if global_path.exists() {
26        let global: Config = load_yaml(&global_path)?;
27        merge_configs(&mut config, global);
28    }
29
30    // Load project config on top (overrides)
31    if project_path.exists() {
32        let project: Config = load_yaml(&project_path)?;
33        merge_configs(&mut config, project);
34    }
35
36    // Agents are defined exclusively as Markdown files with YAML frontmatter.
37    // Load them from the global and project agents/ directories (merged by name).
38    // `session.max_steps` is the default for agents that don't declare it.
39    config.agents = crate::agent::loader::load_agents(config.session.max_steps, explicit_root)?;
40
41    // Expand a leading `~` in the database path to the user's home directory
42    // so we don't create a literal `~` directory in the CWD.
43    config.session.database_path =
44        crate::config::paths::expand_tilde(&config.session.database_path);
45
46    Ok(config)
47}

57/// Load a YAML file and deserialize it, expanding `${VAR}` env vars first.
58fn load_yaml<T: serde::de::DeserializeOwned>(path: &Path) -> Result<T> {
59    let contents = std::fs::read_to_string(path)
60        .map_err(|e| Error::Config(format!("Failed to read {}: {}", path.display(), e)))?;
61    let expanded = expand_env_vars(&contents);
62    let value: T = serde_yaml::from_str(&expanded)
63        .map_err(|e| Error::Config(format!("Failed to parse {}: {}", path.display(), e)))?;
64    Ok(value)
65}

67/// Replace `${VAR_NAME}` patterns with the value of the corresponding environment variable.
68/// Leaves unmatched variables as-is (so missing keys don't crash, they'll fail at auth).
69fn expand_env_vars(s: &str) -> String {
70    let mut result = String::with_capacity(s.len());
71    let mut rest = s;
72    while let Some(dollar) = rest.find("${") {
73        result.push_str(&rest[..dollar]);
74        rest = &rest[dollar + 2..];
75        if let Some(close) = rest.find('}') {
76            let var_name = &rest[..close];
77            let value = std::env::var(var_name).unwrap_or_else(|_| format!("${{{}}}", var_name));
78            result.push_str(&value);
79            rest = &rest[close + 1..];
80        } else {
81            // Unclosed ${ — keep literal
82            result.push_str("${");
83            break;
84        }
85    }
86    result.push_str(rest);
87    result
88}

90/// Merge `override_config` into `base_config`. Project values take precedence.
91pub fn merge_configs(base: &mut Config, override_cfg: Config) {
92    // Merge models
93    if let Some(anthropic) = override_cfg.models.anthropic {
94        base.models.anthropic = Some(anthropic);
95    }
96    if let Some(openai) = override_cfg.models.openai {
97        base.models.openai = Some(openai);
98    }
99    if let Some(openrouter) = override_cfg.models.openrouter {
100        base.models.openrouter = Some(openrouter);
101    }
102    if let Some(ollama) = override_cfg.models.ollama {
103        base.models.ollama = Some(ollama);
104    }
105    if let Some(bedrock) = override_cfg.models.bedrock {
106        base.models.bedrock = Some(bedrock);
107    }
108    if let Some(azure) = override_cfg.models.azure {
109        base.models.azure = Some(azure);
110    }
111    if let Some(google) = override_cfg.models.google {
112        base.models.google = Some(google);
113    }
114    // Merge the prompt-caching policy (project wins over global).
115    if override_cfg.models.cache.mode != crate::config::types::CacheMode::Auto {
116        base.models.cache = override_cfg.models.cache;
117    }
118
119    // Merge MCPs
120    base.mcps.extend(override_cfg.mcps);
121
122    // Merge session
123    if override_cfg.session.history_limit_percent != 50.0 {
124        base.session.history_limit_percent = override_cfg.session.history_limit_percent;
125    }
126    if override_cfg.session.database_path != default_db_path() {
127        base.session.database_path = override_cfg.session.database_path;
128    }
129    if override_cfg.session.max_steps != default_max_steps() {
130        base.session.max_steps = override_cfg.session.max_steps;
131    }
132
133    // Merge keymap overrides (project wins over global).
134    if override_cfg.keymap.is_some() {
135        base.keymap = override_cfg.keymap;
136    }
137
138    // Merge editor override.
139    if override_cfg.editor.is_some() {
140        base.editor = override_cfg.editor;
141    }
142
143    // Merge model picker favorites.
144    if !override_cfg.model_picker.favorites.is_empty() {
145        base.model_picker.favorites = override_cfg.model_picker.favorites;
146    }
147
148    // Merge custom slash commands (project wins over global).
149    if !override_cfg.commands.is_empty() {
150        base.commands = override_cfg.commands;
151    }
152
153    // NOTE: Agents are NOT merged here. They are defined exclusively as
154    // Markdown files and loaded by crate::agent::loader::load_agents().
155}

215    fn test_expand_env_vars() {
216        unsafe { std::env::set_var("ANACLETO_TEST_KEY", "sk-secret-123") };
217        assert_eq!(expand_env_vars("${ANACLETO_TEST_KEY}"), "sk-secret-123");
218        assert_eq!(
219            expand_env_vars("prefix-${ANACLETO_TEST_KEY}-suffix"),
220            "prefix-sk-secret-123-suffix"
221        );
222        // Unmatched vars stay literal
223        assert_eq!(expand_env_vars("${MISSING_VAR_XYZ}"), "${MISSING_VAR_XYZ}");
224        // Plain text passes through
225        assert_eq!(expand_env_vars("no vars here"), "no vars here");
226        unsafe { std::env::remove_var("ANACLETO_TEST_KEY") };
227    }
