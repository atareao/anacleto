4//! commands (`/skills`, `/mcps`, `/status`, `/init`, `/review`, `/warp`,
5//! `/workspaces`, `/move`, `/worktree`, `/timeline`, `/build`, `/parent`,

25    /// Handle `/skills`: list the skills of the root agent.
26    pub(crate) async fn handle_list_skills(&self) -> Result<()> {
27        let skills = self
28            .config
29            .agents
30            .iter()
31            .find(|a| a.role == AgentRole::Root)
32            .map(|c| load_agent_skills(&c.skills))
33            .unwrap_or_default();
34        let infos: Vec<SkillInfo> = skills
35            .iter()
36            .map(|s| SkillInfo {
37                name: s.name.clone(),
38                description: s.description.clone(),
39            })
40            .collect();
41        self.event_tx
42            .send(EngineEvent::SkillsListed(infos))
43            .await
44            .ok();
45        Ok(())
46    }

48    /// Handle `/mcps`: list the MCP servers and their enabled state.
49    pub(crate) async fn handle_list_mcps(&self) -> Result<()> {
50        let enabled_map = self.mcp_enabled.lock().await;
51        let statuses: Vec<McpStatus> = self
52            .mcp_registry
53            .lock()
54            .await
55            .names()
56            .iter()
57            .map(|n| McpStatus {
58                name: n.clone(),
59                enabled: *enabled_map.get(n).unwrap_or(&true),
60            })
61            .collect();
62        drop(enabled_map);
63        self.event_tx
64            .send(EngineEvent::McpsListed(statuses))
65            .await
66            .ok();
67        Ok(())
68    }

81    /// Handle `/status`: produce an engine status report.
82    pub(crate) async fn handle_status(&self) -> Result<()> {
83        let session_id = self.active_session_id;
84        let session_name = match (session_id, &self.database) {
85            (Some(id), Some(db)) => db
86                .get_session_name(id)
87                .await?
88                .unwrap_or_else(|| "unknown".into()),
89            _ => "none".into(),
90        };
91        let provider_name = if self.current_model.contains('/') {
92            "openrouter"
93        } else if self.current_model.starts_with("claude") {
94            "anthropic"
95        } else if self.current_model.starts_with("gpt")
96            || self.current_model.starts_with("o1")
97            || self.current_model.starts_with("o3")
98        {
99            "openai"
100        } else {
101            "ollama"
102        };
103        let context_window = self
104            .llm_registry
105            .get(provider_name)
106            .map(|p| p.context_window() as u32)
107            .unwrap_or(0);
108        let info = StatusInfo {
109            model: self.current_model.clone(),
110            session_id,
111            session_name,
112            total_tokens: self.total_tokens,
113            context_window,
114            cost: self.total_cost,
115            debug: self.debug.load(Ordering::Relaxed),
116            workspace: self.workspace.clone(),
117        };
118        self.event_tx
119            .send(EngineEvent::StatusReport(info))
120            .await
121            .ok();
122        Ok(())
123    }

125    /// Handle `/init`: generate AGENTS.md in the workspace from collected answers.
126    pub(crate) async fn handle_init(&mut self, answers: InitAnswers) -> Result<()> {
127        let mut content = format!(
128            "# {}\n\n{}",
129            answers.name,
130            if answers.description.is_empty() {
131                "# Anacleto agent".to_string()
132            } else {
133                answers.description
134            }
135        );
136        if !answers.stack.trim().is_empty() {
137            content.push_str(&format!("\n\n## Tech stack\n\n{}", answers.stack));
138        }
139        let path = self.workspace.join("AGENTS.md");
140        tokio::fs::write(&path, content).await.map_err(Error::Io)?;
141        self.event_tx.send(EngineEvent::InitDone).await.ok();
142        Ok(())
143    }

145    /// Handle `/review`: run git diff and send it to the root agent for review.
146    pub(crate) async fn handle_review(&mut self, target: Option<String>) -> Result<()> {
147        let mut cmd = std::process::Command::new("git");
148        cmd.arg("diff");
149        if let Some(t) = &target {
150            cmd.arg(t);
151        }
152        cmd.current_dir(&self.workspace);
153        let output = cmd.output().map_err(Error::Io)?;
154        let diff = String::from_utf8_lossy(&output.stdout).to_string();
155        let title = match &target {
156            Some(t) => format!("git diff {}", t),
157            None => "git diff".to_string(),
158        };
159        self.event_tx
160            .send(EngineEvent::DiffAvailable {
161                text: diff.clone(),
162                title,
163            })
164            .await
165            .ok();
166        let prompt = if diff.trim().is_empty() {
167            "No hay cambios sin commitear para revisar.".to_string()
168        } else {
169            format!(
170                "Revisa los siguientes cambios de git:\n\n```diff\n{}\n```",
171                diff
172            )
173        };
174        self.send_to_active(AgentMessage::UserInput { content: prompt })
175            .await?;
176        self.event_tx
177            .send(EngineEvent::ReviewResult(diff))
178            .await
179            .ok();
180        Ok(())
181    }

224            .send(EngineEvent::SessionMoved {
225                session_id,
226                workspace: workspace.to_string(),
227            })
228            .await
229            .ok();
230        Ok(())

290    /// Handle `/build`: read the plan markdown file from the workspace and
291    /// inject it as an execution message to the active agent.
292    ///
293    /// If `PLAN.md` does not exist, a descriptive [`EngineEvent::Error`] is
294    /// emitted (instead of aborting the command) so the user gets clear
295    /// feedback and the engine loop continues.
296    pub(crate) async fn handle_build(&mut self) -> Result<()> {
297        let path = self.workspace.join("PLAN.md");
298        let content = match tokio::fs::read_to_string(&path).await {
299            Ok(c) => c,
300            Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
301                self.event_tx
302                    .send(EngineEvent::Error {
303                        agent_id: None,
304                        message: format!(
305                            "No se encontró PLAN.md en el workspace ({}). \
306                             Crea el plan antes de usar /build.",
307                            path.display()
308                        ),
309                    })
310                    .await
311                    .ok();
312                return Ok(());
313            }
314            Err(e) => return Err(Error::Io(e)),
315        };
316        let prompt = format!(
317            "Execute the following plan. Implement it fully, then report what was done.\n\n{}",
318            content
319        );
320        self.send_to_active(AgentMessage::UserInput { content: prompt })
321            .await?;
322        self.event_tx.send(EngineEvent::BuildDone).await.ok();
323        Ok(())
324    }

363    /// Handle `/snapshot`: create a snapshot of the active session's conversation.
364    ///
365    /// The snapshot captures the serialized message list so it can be restored
366    /// later via `/revert`.
367    pub(crate) async fn handle_snapshot(&mut self, name: Option<&str>) -> Result<()> {
368        let Some(session_id) = self.active_session_id else {
369            return Ok(());
370        };
371        let Some(ref db) = self.database else {
372            return Ok(());
373        };
374        let messages = db.get_session_messages(session_id).await?;
375        let content = serde_json::to_string(&messages)?;
376        let snapshot_name = name.unwrap_or("snapshot").to_string();
377        let snapshot = db
378            .create_snapshot(session_id, &snapshot_name, &content)
379            .await?;
380        self.event_tx
381            .send(EngineEvent::SnapshotCreated {
382                snapshot: snapshot.clone(),
383            })
384            .await
385            .ok();
386        Ok(())
387    }

389    /// Handle `/revert`: restore the active session to a snapshot's state.
390    ///
391    /// The current messages are deleted and replaced with the snapshot's
392    /// serialized message list, then the root agent's context is reloaded.
393    pub(crate) async fn handle_revert(&mut self, snapshot_id: Uuid) -> Result<()> {
394        let Some(session_id) = self.active_session_id else {
395            return Ok(());
396        };
397        let Some(ref db) = self.database else {
398            return Ok(());
399        };
400        let Some(snapshot) = db.get_snapshot(snapshot_id).await? else {
401            return Err(Error::NotFound(format!(
402                "Snapshot '{snapshot_id}' not found"
403            )));
404        };
405        if snapshot.session_id != session_id {
406            return Err(Error::Session(format!(
407                "Snapshot '{snapshot_id}' does not belong to the active session"
408            )));
409        }
410        // Parse the snapshot's messages BEFORE touching the session, so that a
411        // deserialization failure leaves the current messages intact instead of
412        // leaving the session empty/corrupt. The deletion only happens after
413        // the restore payload has been fully loaded and validated in memory.
414        let restored: Vec<StoredMessage> = serde_json::from_str(&snapshot.content)?;
415        // Remove all current messages, then restore the snapshot's messages.
416        let current = db.get_session_messages(session_id).await?;
417        if !current.is_empty() {
418            db.delete_messages(session_id, current.len()).await?;
419        }
420        db.restore_messages(session_id, &restored).await?;
421        self.reload_history_to_root(session_id).await?;
422        self.event_tx
423            .send(EngineEvent::SnapshotReverted { snapshot_id })
424            .await
425            .ok();
426        Ok(())
427    }

445    /// Handle `/stage`: capture the current conversation state as a staged
446    /// snapshot without persisting it. The staged snapshot can be committed
447    /// later via `/commit` or discarded via `/clear`.
448    pub(crate) async fn handle_stage(&mut self, name: Option<&str>) -> Result<()> {
449        let Some(session_id) = self.active_session_id else {
450            return Ok(());
451        };
452        let Some(ref db) = self.database else {
453            return Ok(());
454        };
455        let messages = db.get_session_messages(session_id).await?;
456        let content = serde_json::to_string(&messages)?;
457        let snapshot_name = name.unwrap_or("staged").to_string();
458        let snapshot = db
459            .create_snapshot(session_id, &snapshot_name, &content)
460            .await?;
461        self.staged_snapshot = Some(snapshot.clone());
462        self.event_tx
463            .send(EngineEvent::SnapshotCreated { snapshot })
464            .await
465            .ok();
466        Ok(())
467    }

469    /// Handle `/clear`: discard the staged snapshot.
470    pub(crate) async fn handle_clear(&mut self) -> Result<()> {
471        if let Some(staged) = self.staged_snapshot.take() {
472            if let Some(ref db) = self.database {
473                db.delete_snapshot(staged.id).await?;
474            }
475        }
476        Ok(())
477    }

479    /// Handle `/commit`: persist the staged snapshot as a named snapshot.
480    ///
481    /// The staged snapshot is renamed (if a name is provided) and kept; the
482    /// staging slot is cleared.
483    pub(crate) async fn handle_commit(&mut self, name: Option<&str>) -> Result<()> {
484        let Some(staged) = self.staged_snapshot.take() else {
485            return Err(Error::Session(
486                "No staged snapshot to commit. Use /stage first.".into(),
487            ));
488        };
489        if let Some(ref db) = self.database {
490            if let Some(new_name) = name {
491                // Rename by re-creating with the same content and deleting the old.
492                let content = staged.content.clone();
493                let renamed = db
494                    .create_snapshot(staged.session_id, new_name, &content)
495                    .await?;
496                db.delete_snapshot(staged.id).await?;
497                self.event_tx
498                    .send(EngineEvent::SnapshotCreated { snapshot: renamed })
499                    .await
500                    .ok();
501            } else {
502                self.event_tx
503                    .send(EngineEvent::SnapshotCreated { snapshot: staged })
504                    .await
505                    .ok();
506            }
507        }
508        Ok(())
509    }

511    /// Handle approval response from the TUI.
512    pub(crate) async fn handle_approval_response(&self, id: &str, approved: bool) {
513        let mut pending = self.pending_approvals.lock().await;
514        if let Some(sender) = pending.remove(id) {
515            let _ = sender.send(approved);
516        }
