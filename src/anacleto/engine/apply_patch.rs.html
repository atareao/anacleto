1//! Batch file patching for the `apply_patch` tool.
2//!
3//! The LLM sends a JSON batch of operations (`add` / `update` / `delete`) that
4//! are applied to files relative to the agent's workspace. All operations are
5//! validated up-front (path traversal is rejected) and applied sequentially
6//! while preserving each existing file's encoding (UTF-8 BOM and CRLF vs LF).
7
8use serde::{Deserialize, Serialize};
9use std::path::{Component, Path, PathBuf};
10
11/// The kind of a single patch operation.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
13#[serde(rename_all = "lowercase")]
14pub enum PatchOpKind {
15    /// Create a new file (creating parent directories as needed).
16    Add,
17    /// Replace the contents of an existing file.
18    Update,
19    /// Delete an existing file.
20    Delete,
21}
22
23/// A single patch operation.
24#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
25pub struct PatchOp {
26    /// The operation to perform.
27    pub op: PatchOpKind,
28    /// File path relative to the workspace.
29    pub path: String,
30    /// New file contents (required for `add`/`update`, ignored for `delete`).
31    #[serde(default)]
32    pub content: Option<String>,
33}
34
35/// A batch of patch operations as sent by the model.
36#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
37pub struct PatchBatch {
38    /// The operations to apply, in order.
39    pub operations: Vec<PatchOp>,
40}
41
42/// The detected encoding of an existing file.
43#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
44pub struct FileEncoding {
45    /// Whether the file starts with a UTF-8 BOM (`EF BB BF`).
46    pub has_bom: bool,
47    /// Whether the file uses CRLF (`\r\n`) line endings.
48    pub crlf: bool,
49}
50
51/// Parse a patch batch from the raw JSON arguments of a tool call.
52///
53/// Accepts either an object with an `operations` array or a bare array of
54/// operations, for robustness.
55pub fn parse_patch_batch(json: &str) -> Result<PatchBatch, String> {
56    let value: serde_json::Value =
57        serde_json::from_str(json).map_err(|e| format!("Invalid patch JSON: {e}"))?;
58
59    let ops = match &value {
60        serde_json::Value::Array(arr) => arr.clone(),
61        serde_json::Value::Object(map) => map
62            .get("operations")
63            .and_then(|v| v.as_array())
64            .cloned()
65            .ok_or_else(|| "Patch batch must contain an 'operations' array".to_string())?,
66        _ => {
67            return Err(
68                "Patch batch must be an object with 'operations' or a bare array".to_string(),
69            );
70        }
71    };
72
73    let operations: Vec<PatchOp> = ops
74        .into_iter()
75        .map(|v| serde_json::from_value(v).map_err(|e| format!("Invalid patch operation: {e}")))
76        .collect::<Result<_, _>>()?;
77
78    if operations.is_empty() {
79        return Err("Patch batch contains no operations".to_string());
80    }
81
82    Ok(PatchBatch { operations })
83}

85/// Resolve a workspace-relative path, rejecting any path that escapes the
86/// workspace (absolute paths, `..` traversal, or symlink escapes).
87pub fn resolve_within_workspace(workspace: &Path, rel: &str) -> Result<PathBuf, String> {
88    let rel_path = Path::new(rel);
89
90    if rel_path.is_absolute() {
91        return Err(format!("Absolute paths are not allowed: {rel}"));
92    }
93
94    for comp in rel_path.components() {
95        match comp {
96            Component::ParentDir => {
97                return Err(format!("Path escapes workspace: {rel}"));
98            }
99            Component::RootDir | Component::Prefix(_) => {
100                return Err(format!("Path escapes workspace: {rel}"));
101            }
102            Component::CurDir | Component::Normal(_) => {}
103        }
104    }
105
106    let workspace_canon = workspace
107        .canonicalize()
108        .map_err(|e| format!("Invalid workspace '{}': {e}", workspace.display()))?;
109
110    let full = workspace_canon.join(rel_path);
111
112    // Canonicalize the nearest existing ancestor of the parent directory to
113    // resolve any symlinks or `..` that could otherwise escape the workspace,
114    // then verify containment. The immediate parent may not exist yet (e.g. an
115    // "add" operation that creates nested directories), so we walk up to the
116    // closest existing ancestor before canonicalizing.
117    let parent = full.parent().unwrap_or(&workspace_canon);
118    let mut probe = parent.to_path_buf();
119    while !probe.exists() {
120        match probe.parent() {
121            Some(p) => probe = p.to_path_buf(),
122            None => break,
123        }
124    }
125    let anchor = if probe.exists() {
126        probe
127            .canonicalize()
128            .map_err(|e| format!("Invalid parent path for '{rel}': {e}"))?
129    } else {
130        workspace_canon.clone()
131    };
132
133    if !anchor.starts_with(&workspace_canon) {
134        return Err(format!("Path escapes workspace: {rel}"));
135    }
136
137    Ok(full)
138}

140/// Detect the encoding (BOM + line endings) of an existing file's bytes.
141pub fn detect_encoding(bytes: &[u8]) -> FileEncoding {
142    let has_bom = bytes.starts_with(&[0xEF, 0xBB, 0xBF]);
143    let crlf = bytes.windows(2).any(|w| w == b"\r\n");
144    FileEncoding { has_bom, crlf }
145}
146
147/// Encode new content using a detected file encoding.
148///
149/// Re-adds the UTF-8 BOM if the original file had one, and converts LF line
150/// endings to CRLF if the original file used CRLF. New files use the default
151/// encoding (no BOM, LF).
152pub fn encode_content(content: &str, enc: &FileEncoding) -> Vec<u8> {
153    let mut out = Vec::new();
154    if enc.has_bom {
155        out.extend_from_slice(&[0xEF, 0xBB, 0xBF]);
156    }
157    if enc.crlf {
158        // Normalize any existing CRLF first so we never double the `\r`.
159        let normalized = content.replace("\r\n", "\n");
160        let with_crlf = normalized.replace('\n', "\r\n");
161        out.extend_from_slice(with_crlf.as_bytes());
162    } else {
163        out.extend_from_slice(content.as_bytes());
164    }
165    out
166}

168/// Resolve a patch path.
169///
170/// When `allow_external` is `false`, paths must stay within the workspace
171/// (absolute paths and `..` traversal are rejected). When `allow_external` is
172/// `true`, absolute paths and traversal are permitted, so the caller must have
173/// already verified the `fs.external` permission.
174pub fn resolve_patch_path(
175    workspace: &Path,
176    rel: &str,
177    allow_external: bool,
178) -> Result<PathBuf, String> {
179    if allow_external {
180        let p = Path::new(rel);
181        if p.is_absolute() {
182            Ok(p.to_path_buf())
183        } else {
184            Ok(workspace.join(p))
185        }
186    } else {
187        resolve_within_workspace(workspace, rel)
188    }
189}

191/// Apply a batch of patch operations to the workspace.
192///
193/// All paths are validated before any change is made; if any path escapes the
194/// workspace the whole batch is rejected without touching the filesystem.
195/// When `allow_external` is `true`, paths may escape the workspace (the caller
196/// is responsible for having granted the `fs.external` permission).
197pub fn apply_patch_batch(
198    workspace: &Path,
199    batch: &PatchBatch,
200    allow_external: bool,
201) -> Result<Vec<String>, String> {
202    // Validate every path up-front so a bad path aborts the whole batch before
203    // any file is modified.
204    let mut resolved = Vec::with_capacity(batch.operations.len());
205    for op in &batch.operations {
206        let full = resolve_patch_path(workspace, &op.path, allow_external)?;
207        resolved.push(full);
208    }
209
210    let mut results = Vec::with_capacity(batch.operations.len());
211    for (op, full) in batch.operations.iter().zip(resolved.iter()) {
212        match op.op {
213            PatchOpKind::Add => {
214                if full.exists() {
215                    return Err(format!("Cannot add, file already exists: {}", op.path));
216                }
217                if let Some(parent) = full.parent() {
218                    std::fs::create_dir_all(parent).map_err(|e| {
219                        format!("Failed to create directories for '{}': {e}", op.path)
220                    })?;
221                }
222                let content = op.content.as_deref().unwrap_or("");
223                // New files use the default encoding (no BOM, LF).
224                std::fs::write(full, content.as_bytes())
225                    .map_err(|e| format!("Failed to write '{}': {e}", op.path))?;
226                results.push(format!("Added {}", op.path));
227            }
228            PatchOpKind::Update => {
229                if !full.exists() {
230                    return Err(format!("Cannot update, file does not exist: {}", op.path));
231                }
232                let bytes = std::fs::read(full)
233                    .map_err(|e| format!("Failed to read '{}': {e}", op.path))?;
234                let enc = detect_encoding(&bytes);
235                let content = op.content.as_deref().unwrap_or("");
236                let encoded = encode_content(content, &enc);
237                std::fs::write(full, encoded)
238                    .map_err(|e| format!("Failed to write '{}': {e}", op.path))?;
239                results.push(format!("Updated {}", op.path));
240            }
241            PatchOpKind::Delete => {
242                if !full.exists() {
243                    return Err(format!("Cannot delete, file does not exist: {}", op.path));
244                }
245                std::fs::remove_file(full)
246                    .map_err(|e| format!("Failed to delete '{}': {e}", op.path))?;
247                results.push(format!("Deleted {}", op.path));
248            }
249        }
250    }
251
252    Ok(results)
253}

255/// Build a unified-diff-style text representation of a patch batch, suitable
256/// for display in the TUI diff viewer. Because `apply_patch` operates on whole
257/// file contents (not line hunks), added/updated files are shown as full
258/// additions and deleted files as a removal header.
259pub fn batch_to_unified_diff(batch: &PatchBatch) -> String {
260    let mut out = String::new();
261    for op in &batch.operations {
262        out.push_str(&format!("--- a/{}\n", op.path));
263        out.push_str(&format!("+++ b/{}\n", op.path));
264        match op.op {
265            PatchOpKind::Add | PatchOpKind::Update => {
266                let content = op.content.as_deref().unwrap_or("");
267                let lines: Vec<&str> = content.lines().collect();
268                let count = lines.len();
269                out.push_str(&format!("@@ -0,0 +1,{} @@\n", count));
270                for line in lines {
271                    out.push_str(&format!("+{}\n", line));
272                }
273            }
274            PatchOpKind::Delete => {
275                out.push_str("@@ -1,0 +0,0 @@\n");
276                out.push_str("-<file deleted>\n");
277            }
278        }
279    }
280    out
