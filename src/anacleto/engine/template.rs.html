1//! Variable expansion for custom slash command templates.
2//!
3//! Custom commands defined in config may contain `{env:VAR}` and `{file:path}`
4//! placeholders that are expanded at dispatch time:
5//!
6//! - `{env:VAR}` — replaced by the value of the environment variable `VAR`.
7//!   If the variable is unset, the placeholder is left literal.
8//! - `{file:path}` — replaced by the contents of the file at `path` (relative
9//!   to the current working directory). If the file cannot be read, the
10//!   placeholder is left literal.
11
12use std::collections::HashMap;
13
14/// Expand `{env:VAR}` and `{file:path}` placeholders in `template`.
15///
16/// Unknown placeholders and unreadable files/env vars are left literal.
17pub fn expand_vars(template: &str, env: &HashMap<String, String>) -> String {
18    let mut result = String::with_capacity(template.len());
19    let mut rest = template;
20
21    while let Some(start) = rest.find('{') {
22        result.push_str(&rest[..start]);
23        let after = &rest[start + 1..];
24        // Find the matching closing brace.
25        let Some(end_rel) = after.find('}') else {
26            // Unclosed brace — keep the remainder literal.
27            result.push_str(&rest[start..]);
28            return result;
29        };
30        let inner = &after[..end_rel];
31        let placeholder = &rest[start..=start + 1 + end_rel];
32
33        if let Some(var) = inner.strip_prefix("env:") {
34            match env.get(var) {
35                Some(value) => result.push_str(value),
36                None => result.push_str(placeholder),
37            }
38        } else if let Some(path) = inner.strip_prefix("file:") {
39            match std::fs::read_to_string(path) {
40                Ok(contents) => result.push_str(contents.trim_end()),
41                Err(_) => result.push_str(placeholder),
42            }
43        } else {
44            // Unknown placeholder — keep literal.
45            result.push_str(placeholder);
46        }
47
48        rest = &after[end_rel + 1..];
49    }
50
51    result.push_str(rest);
52    result
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58
59    fn env() -> HashMap<String, String> {
60        let mut m = HashMap::new();
61        m.insert("USER".to_string(), "lorenzo".to_string());
62        m
63    }
64
65    #[test]
66    fn test_expand_env_var() {
67        assert_eq!(expand_vars("hello {env:USER}", &env()), "hello lorenzo");
68    }
69
70    #[test]
71    fn test_expand_env_var_missing_stays_literal() {
72        assert_eq!(
73            expand_vars("hello {env:MISSING}", &env()),
74            "hello {env:MISSING}"
75        );
76    }
77
78    #[test]
79    fn test_expand_file_var() {
80        let tmp = tempfile::tempdir().unwrap();
81        let path = tmp.path().join("note.txt");
82        std::fs::write(&path, "file contents\n").unwrap();
83        let tpl = format!("read: {{file:{}}}", path.display());
84        assert_eq!(expand_vars(&tpl, &env()), "read: file contents");
85    }
86
87    #[test]
88    fn test_expand_file_missing_stays_literal() {
89        let tpl = "read: {file:/nonexistent/xyz.txt}";
90        assert_eq!(expand_vars(tpl, &env()), tpl);
91    }
92
93    #[test]
94    fn test_unknown_placeholder_stays_literal() {
95        assert_eq!(expand_vars("a {foo} b", &env()), "a {foo} b");
96    }
97
98    #[test]
99    fn test_unclosed_brace_stays_literal() {
100        assert_eq!(expand_vars("a {env:USER", &env()), "a {env:USER");
101    }
102
103    #[test]
104    fn test_multiple_placeholders() {
105        assert_eq!(
106            expand_vars("{env:USER}-{env:USER}", &env()),
107            "lorenzo-lorenzo"
108        );
109    }
110
111    #[test]
112    fn test_plain_text_passthrough() {
113        assert_eq!(expand_vars("no vars", &env()), "no vars");
114    }
115}
