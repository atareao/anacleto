12/// A connected MCP client.
13pub struct McpClient {
14    /// Server name.
15    name: McpName,
16    /// Transport configuration.
17    transport: McpTransport,
18    /// Child process (for stdio transport).
19    child: Option<Mutex<Child>>,
20    /// Server info (populated after initialize).
21    info: Option<McpServerInfo>,
22    /// Next request ID.
23    next_id: u64,
24    /// Stdin handle for sending requests (stdio transport).
25    stdin: Option<tokio::process::ChildStdin>,
26    /// Stdout handle for reading responses (stdio transport).
27    stdout: Option<tokio::io::BufReader<tokio::process::ChildStdout>>,
28    /// TCP stream (tcp transport).
29    tcp_stream: Option<TcpStream>,
30}

32impl McpClient {
33    /// Create a new MCP client from a definition.
34    pub fn new(name: McpName, definition: &McpDefinition) -> Self {
35        Self {
36            name,
37            transport: McpTransport::from(definition),
38            child: None,
39            info: None,
40            next_id: 1,
41            stdin: None,
42            stdout: None,
43            tcp_stream: None,
44        }
45    }

47    /// Connect to the MCP server and perform handshake.
48    pub async fn connect(&mut self) -> Result<McpServerInfo> {
49        let transport = self.transport.clone();
50        match &transport {
51            McpTransport::Stdio { command, args } => self.connect_stdio(command, args).await,
52            McpTransport::Tcp { host, port } => self.connect_tcp(host, *port).await,
53        }
54    }
55
56    async fn connect_stdio(&mut self, command: &str, args: &[String]) -> Result<McpServerInfo> {
57        let mut child = Command::new(command)
58            .args(args)
59            .stdin(std::process::Stdio::piped())
60            .stdout(std::process::Stdio::piped())
61            .stderr(std::process::Stdio::piped())
62            .spawn()
63            .map_err(|e| Error::Mcp(format!("Failed to spawn MCP server: {}", e)))?;
64
65        let stdin = child
66            .stdin
67            .take()
68            .ok_or_else(|| Error::Mcp("Failed to capture stdin".into()))?;
69        let stdout = child
70            .stdout
71            .take()
72            .ok_or_else(|| Error::Mcp("Failed to capture stdout".into()))?;
73        let reader = tokio::io::BufReader::new(stdout);
74
75        self.stdin = Some(stdin);
76        self.stdout = Some(reader);
77
78        // Perform initialize handshake
79        let info = self.perform_initialize().await?;
80
81        self.child = Some(Mutex::new(child));
82        self.info = Some(info.clone());
83        Ok(info)
84    }
85
86    async fn connect_tcp(&mut self, host: &str, port: u16) -> Result<McpServerInfo> {
87        let addr = format!("{}:{}", host, port);
88        let stream = TcpStream::connect(&addr).await.map_err(|e| {
89            Error::Mcp(format!(
90                "Failed to connect to TCP MCP server at {}: {}",
91                addr, e
92            ))
93        })?;
94        self.tcp_stream = Some(stream);
95        let info = self.perform_initialize().await?;
96        self.info = Some(info.clone());
97        Ok(info)
98    }
99
100    async fn perform_initialize(&mut self) -> Result<McpServerInfo> {
101        // Send initialize request
102        let request = serde_json::json!({
103            "jsonrpc": "2.0",
104            "id": self.next_id,
105            "method": "initialize",
106            "params": {
107                "protocolVersion": "2024-11-05",
108                "capabilities": {},
109                "clientInfo": {
110                    "name": "anacleto",
111                    "version": "0.1.0"
112                }
113            }
114        });
115        self.next_id += 1;
116
117        self.send_jsonrpc(&request).await?;
118        let response = self.read_jsonrpc().await?;
119
120        // Parse server info from response
121        let result = response
122            .get("result")
123            .ok_or_else(|| Error::Mcp("Initialize response missing 'result'".into()))?;
124
125        let server_name = result
126            .get("serverInfo")
127            .and_then(|s| s.get("name"))
128            .and_then(|s| s.as_str())
129            .unwrap_or(&self.name)
130            .to_string();
131        let server_version = result
132            .get("serverInfo")
133            .and_then(|s| s.get("version"))
134            .and_then(|s| s.as_str())
135            .unwrap_or("0.1.0")
136            .to_string();
137        let capabilities = result.get("capabilities").cloned().unwrap_or_default();
138
139        // Send initialized notification
140        let notification = serde_json::json!({
141            "jsonrpc": "2.0",
142            "method": "notifications/initialized",
143            "params": {}
144        });
145        self.send_jsonrpc(&notification).await?;
146
147        // List tools if supported
148        let tools = if capabilities
149            .get("tools")
150            .and_then(|t| t.as_bool())
151            .unwrap_or(false)
152        {
153            self.list_tools_inner().await.unwrap_or_default()
154        } else {
155            vec![]
156        };
157
158        Ok(McpServerInfo {
159            name: server_name,
160            version: server_version,
161            capabilities: McpCapabilities {
162                tools: capabilities
163                    .get("tools")
164                    .and_then(|t| t.as_bool())
165                    .unwrap_or(false),
166                resources: capabilities
167                    .get("resources")
168                    .and_then(|t| t.as_bool())
169                    .unwrap_or(false),
170                prompts: capabilities
171                    .get("prompts")
172                    .and_then(|t| t.as_bool())
173                    .unwrap_or(false),
174            },
175            tools,
176            resources: vec![],
177        })
178    }

180    /// Call a tool on the MCP server.
181    pub async fn call_tool(&mut self, call: McpToolCall) -> Result<McpToolResult> {
182        let request = serde_json::json!({
183            "jsonrpc": "2.0",
184            "id": self.next_id,
185            "method": "tools/call",
186            "params": {
187                "name": call.name,
188                "arguments": call.arguments
189            }
190        });
191        self.next_id += 1;
192
193        self.send_jsonrpc(&request).await?;
194        let response = self.read_jsonrpc().await?;
195
196        if let Some(error) = response.get("error") {
197            return Ok(McpToolResult {
198                success: false,
199                content: serde_json::Value::Null,
200                error: Some(
201                    error
202                        .get("message")
203                        .and_then(|m| m.as_str())
204                        .unwrap_or("Unknown error")
205                        .to_string(),
206                ),
207            });
208        }
209
210        let result = response.get("result").cloned().unwrap_or_default();
211        let content = result
212            .get("content")
213            .cloned()
214            .unwrap_or(serde_json::Value::Null);
215
216        Ok(McpToolResult {
217            success: true,
218            content,
219            error: None,
220        })
221    }

228    /// Internal: send tools/list request and parse response.
229    async fn list_tools_inner(&mut self) -> Result<Vec<McpTool>> {
230        let request = serde_json::json!({
231            "jsonrpc": "2.0",
232            "id": self.next_id,
233            "method": "tools/list",
234            "params": {}
235        });
236        self.next_id += 1;
237
238        self.send_jsonrpc(&request).await?;
239        let response = self.read_jsonrpc().await?;
240
241        let result = response
242            .get("result")
243            .ok_or_else(|| Error::Mcp("tools/list response missing 'result'".into()))?;
244
245        let tools = result
246            .get("tools")
247            .and_then(|t| serde_json::from_value::<Vec<McpTool>>(t.clone()).ok())
248            .unwrap_or_default();
249
250        Ok(tools)
251    }

253    /// Send a JSON-RPC message over the active transport (stdio or TCP).
254    async fn send_jsonrpc(&mut self, message: &serde_json::Value) -> Result<()> {
255        let data = serde_json::to_string(message)
256            .map_err(|e| Error::Mcp(format!("Failed to serialize JSON-RPC message: {}", e)))?;
257        let data = format!("{}\n", data);
258
259        if let Some(stdin) = self.stdin.as_mut() {
260            stdin
261                .write_all(data.as_bytes())
262                .await
263                .map_err(|e| Error::Mcp(format!("Failed to write to MCP stdin: {}", e)))?;
264        } else if let Some(stream) = self.tcp_stream.as_mut() {
265            stream
266                .write_all(data.as_bytes())
267                .await
268                .map_err(|e| Error::Mcp(format!("Failed to write to MCP TCP stream: {}", e)))?;
269        }
270        Ok(())
271    }
272
273    /// Read a JSON-RPC response line from the active transport (stdio or TCP).
274    async fn read_jsonrpc(&mut self) -> Result<serde_json::Value> {
275        let mut line = String::new();
276
277        if let Some(ref mut reader) = self.stdout {
278            reader
279                .read_line(&mut line)
280                .await
281                .map_err(|e| Error::Mcp(format!("Failed to read from MCP stdout: {}", e)))?;
282        } else if let Some(ref mut stream) = self.tcp_stream {
283            // Read byte by byte until newline
284            loop {
285                let mut byte = [0u8; 1];
286                let n = stream.read(&mut byte).await.map_err(|e| {
287                    Error::Mcp(format!("Failed to read from MCP TCP stream: {}", e))
288                })?;
289                if n == 0 {
290                    break; // EOF
291                }
292                let c = byte[0] as char;
293                line.push(c);
294                if c == '\n' {
295                    break;
296                }
297            }
298        }
299
300        if line.trim().is_empty() {
301            return Ok(serde_json::json!({}));
302        }
303
304        serde_json::from_str(&line)
305            .map_err(|e| Error::Mcp(format!("Failed to parse JSON-RPC response: {}", e)))
306    }

308    /// Disconnect from the MCP server.
309    pub async fn disconnect(&mut self) -> Result<()> {
310        if let Some(child) = self.child.take() {
311            let mut child = child.into_inner();
312            child.kill().await.ok();
313            child.wait().await.ok();
314        }
315        if let Some(stream) = self.tcp_stream.take() {
316            drop(stream);
317        }
318        Ok(())
319    }

327impl McpRegistry {

335    /// Register and connect an MCP server.
336    pub async fn register(&mut self, name: McpName, definition: &McpDefinition) -> Result<()> {
337        let mut client = McpClient::new(name.clone(), definition);
338        client.connect().await?;
339        self.clients.insert(name, Mutex::new(client));
340        Ok(())
341    }
342
343    /// Get a client by name.
344    pub fn get(&self, name: &str) -> Option<&Mutex<McpClient>> {
345        self.clients.get(name)
346    }

348    /// Collect tools from specific MCP servers and convert to ToolDefinitions.
349    /// Returns a list of (server_name, original_tool_name, ToolDefinition) tuples.
350    pub async fn collect_tools(
351        &self,
352        server_names: &[String],
353    ) -> Vec<(String, String, crate::llm::types::ToolDefinition)> {
354        let mut tools = Vec::new();
355        for name in server_names {
356            if let Some(client_lock) = self.clients.get(name) {
357                let mut client = client_lock.lock().await;
358                if let Ok(mcp_tools) = client.list_tools().await {
359                    for t in mcp_tools {
360                        let original_name = t.name.clone();
361                        let tool_def = crate::llm::types::ToolDefinition {
362                            name: format!("{}_{}", name, t.name),
363                            description: format!("[MCP {}] {}", name, t.description),
364                            input_schema: t.input_schema,
365                        };
366                        tools.push((name.clone(), original_name, tool_def));
367                    }
368                }
369            }
370        }
371        tools
372    }
373
374    /// Execute an MCP tool call.
375    pub async fn call_tool(
376        &self,
377        server_name: &str,
378        tool_name: &str,
379        arguments: serde_json::Value,
380    ) -> Result<String> {
381        let client_lock = self
382            .clients
383            .get(server_name)
384            .ok_or_else(|| Error::Mcp(format!("MCP server '{}' not found", server_name)))?;
385
386        let mut client = client_lock.lock().await;
387        let result = client
388            .call_tool(McpToolCall {
389                name: tool_name.to_string(),
390                arguments,
391            })
392            .await?;
393
394        if result.success {
395            Ok(serde_json::to_string_pretty(&result.content).unwrap_or_else(|_| "{}".to_string()))
396        } else {
397            Err(Error::Mcp(
398                result.error.unwrap_or_else(|| "Unknown error".into()),
399            ))
400        }
401    }
402
403    /// Disconnect all clients.
404    pub async fn disconnect_all(&mut self) {
405        for client in self.clients.values_mut() {
406            client.get_mut().disconnect().await.ok();
407        }
408        self.clients.clear();
409    }
