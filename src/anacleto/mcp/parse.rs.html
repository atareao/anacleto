5/// Parse a `resources/list` JSON-RPC response into a list of resources.
6pub(crate) fn parse_resources_response(response: &serde_json::Value) -> Result<Vec<McpResource>> {
7    if let Some(error) = response.get("error") {
8        return Err(Error::Mcp(
9            error
10                .get("message")
11                .and_then(|m| m.as_str())
12                .unwrap_or("Unknown error")
13                .to_string(),
14        ));
15    }
16
17    let result = response
18        .get("result")
19        .ok_or_else(|| Error::Mcp("resources/list response missing 'result'".into()))?;
20
21    let resources = result
22        .get("resources")
23        .and_then(|r| serde_json::from_value::<Vec<McpResource>>(r.clone()).ok())
24        .unwrap_or_default();
25
26    Ok(resources)
27}

55/// Parse a `resources/read` JSON-RPC response into a string.
56///
57/// Text contents are concatenated; binary `blob` contents are exposed as
58/// `data:<mime>;base64,<payload>` data URIs.
59pub(crate) fn parse_read_resource_response(response: &serde_json::Value) -> Result<String> {
60    if let Some(error) = response.get("error") {
61        return Err(Error::Mcp(
62            error
63                .get("message")
64                .and_then(|m| m.as_str())
65                .unwrap_or("Unknown error")
66                .to_string(),
67        ));
68    }
69
70    let result = response
71        .get("result")
72        .ok_or_else(|| Error::Mcp("resources/read response missing 'result'".into()))?;
73
74    let contents = result
75        .get("contents")
76        .and_then(|c| c.as_array())
77        .cloned()
78        .unwrap_or_default();
79
80    if contents.is_empty() {
81        return Ok(String::new());
82    }
83
84    let mut out = String::new();
85    for item in contents {
86        let mime = item
87            .get("mimeType")
88            .and_then(|m| m.as_str())
89            .unwrap_or("text/plain");
90        if let Some(text) = item.get("text").and_then(|t| t.as_str()) {
91            out.push_str(text);
92        } else if let Some(blob) = item.get("blob").and_then(|b| b.as_str()) {
93            // Binary content: expose as a data URI with mime metadata.
94            out.push_str(&format!("data:{mime};base64,{blob}"));
95        }
96        out.push('\n');
97    }
98
99    Ok(out)
100}

106    #[test]
107    fn test_parse_resources_response() {
108        let response = serde_json::json!({
109            "jsonrpc": "2.0",
110            "id": 1,
111            "result": {
112                "resources": [
113                    {
114                        "uri": "file:///tmp/a.txt",
115                        "name": "a.txt",
116                        "description": "A file",
117                        "mimeType": "text/plain"
118                    },
119                    {
120                        "uri": "file:///tmp/b.png",
121                        "name": "b.png"
122                    }
123                ]
124            }
125        });
126        let resources = parse_resources_response(&response).unwrap();
127        assert_eq!(resources.len(), 2);
128        assert_eq!(resources[0].uri, "file:///tmp/a.txt");
129        assert_eq!(resources[0].name, "a.txt");
130        assert_eq!(resources[0].description.as_deref(), Some("A file"));
131        assert_eq!(resources[0].mime_type.as_deref(), Some("text/plain"));
132        // Optional fields default to None.
133        assert_eq!(resources[1].description, None);
134        assert_eq!(resources[1].mime_type, None);
135    }
